    46→/* ═══════════════════════════════════════════════════════════════
    47→   Reaction Club · notes reader — one enclosed room per spec point.
    48→   Each section is a full-width tonal band: most borrow a candy
    49→   colour, every third sits on the deep roast (a dark room), and
    50→   the footer's wave rides between the bands, so the note reads as
    51→   a string of connected rooms rather than one long scroll.
    52→   ═══════════════════════════════════════════════════════════════ */

    59→function BandWave({ from, to }: { from: string; to: string }) {

    73→/* ————— main reader ————— */
    74→
    75→export function NotesReader({ doc, unit, onBack }: { doc: NotesDoc; unit: Unit; onBack: () => void }) {
    76→  const main = unit.candy.main;
    77→  const [active, setActive] = useState<string>(doc.sections[0]?.spec ?? "");
    78→
    79→  const { scrollYProgress } = useScroll();
    80→  const scaleX = useSpring(scrollYProgress, { stiffness: 140, damping: 26, mass: 0.4 });
    81→
    82→  /* scroll spy */
    83→  const spy = useRef<IntersectionObserver | null>(null);
    84→  useEffect(() => {
    85→    spy.current = new IntersectionObserver(
    86→      (entries) => {
    87→        for (const e of entries) {
    88→          if (e.isIntersecting) setActive(e.target.id.replace("spec-", "").replace("-", "."));
    89→        }
    90→      },
    91→      { rootMargin: "-30% 0px -60% 0px" }
    92→    );
    93→    for (const s of doc.sections) {
    94→      const el = document.getElementById(`spec-${s.spec.replace(".", "-")}`);
    95→      if (el) spy.current?.observe(el);
    96→    }
    97→    return () => spy.current?.disconnect();
    98→  }, [doc]);
    99→
   100→  const count = (k: NotesBlock["k"]) => {
   101→    let n = 0;
   102→    const walk = (blocks: NotesBlock[]) => blocks.forEach((b) => b.k === k && n++);
   103→    walk(doc.intro);
   104→    doc.sections.forEach((s) => walk(s.blocks));
   105→    if (doc.outro) walk(doc.outro);
   106→    return n;
   107→  };
   108→  const stats = [
   109→    `${doc.sections.length} spec sections`,
   110→    `${count("worked")} worked examples`,
   111→    `${count("figure")} diagrams`,
   112→    `${count("checkpoint")} textbook checkpoints`,
   113→  ];
   114→
   115→  const accentOf = (i: number) => BANDS[i % BANDS.length];
   116→  const bandOf = (i: number) => tint(accentOf(i), 10);
   117→  /* every third room sits on the deep roast — a dark interlude
   118→     between the candy bands (2.2, 2.5, 2.8 …), fonts flipped to milk */
   119→  const isDarkBand = (i: number) => i % 3 === 1;
   120→  const bgOf = (i: number) => (isDarkBand(i) ? "var(--color-cocoa)" : bandOf(i));
   121→  const activeIndex = Math.max(0, doc.sections.findIndex((s) => s.spec === active));
   122→  const activeSection = doc.sections[activeIndex];
   123→
   124→  return (
   125→    <div className="relative">
   126→      {/* reading progress */}
   127→      <motion.div
   128→        aria-hidden
   129→        className="fixed top-0 left-0 z-[70] h-[7px] w-full origin-left rounded-r-full"
   130→        style={{ scaleX, background: `linear-gradient(90deg, ${main}, #FFD43B)` }}
   131→      />
   132→
   133→      {/* ————— header ————— */}
   134→      <header className="mx-auto max-w-3xl px-4 pt-24 sm:px-8">
   135→        <PopIn>
   136→          <SoftButton onClick={onBack} className="min-h-[42px] px-4 py-2 text-[13.5px]" ariaLabel="Back to the topic page">
   137→            <ArrowLeft className="h-4 w-4" /> Back to the topic page
   138→          </SoftButton>
   139→        </PopIn>
   140→
   141→        <PopIn delay={0.05} className="mt-7 flex flex-wrap items-center gap-2">
   142→          <span
   143→            className="inline-flex -rotate-1 items-center gap-1.5 rounded-full px-3.5 py-1.5 font-display text-[11.5px] font-bold tracking-[0.14em] text-white uppercase"
   144→            style={{ background: main }}
   145→          >
   146→            <BookOpen className="h-3.5 w-3.5" /> Revision note
   147→          </span>
   148→          <span className="inline-flex items-center rounded-full bg-ink/5 px-3.5 py-1.5 text-[12.5px] font-bold text-ink-soft">
   149→            {doc.source}
   150→          </span>
   151→        </PopIn>
   152→        <PopIn delay={0.1}>
   153→          <p className="mt-6 font-display text-[15px] font-semibold tracking-[0.06em] uppercase" style={{ color: shade(main, 62) }}>
   154→            {doc.kicker}
   155→          </p>
   156→          <h1 className="mt-2 font-display text-[clamp(2.1rem,5vw,3.2rem)] leading-[1.05] font-semibold tracking-tight text-ink">
   157→            {doc.title}
   158→          </h1>
   159→        </PopIn>
   160→
   161→        <div className="mt-6 flex flex-col items-start gap-5 sm:flex-row sm:items-center">
   162→          <div className="shrink-0">
   163→            <Mascot main={main} size={104} bubbles={false} />
   164→          </div>
   165→          <Reveal delay={0.15} className="min-w-0">
   166→            <SpeechBubble tailClass="left-8 -top-[7px] rotate-[225deg]">
   167→              <p className="text-[14.5px] leading-[1.75] text-ink">
   168→                {doc.intro.map((b, i) =>
   169→                  b.k === "p" ? (
   170→                    <span key={i} className="block">
   171→                      <Fmt text={b.text} />
   172→                    </span>
   173→                  ) : null
   174→                )}
   175→              </p>
   176→            </SpeechBubble>
   177→          </Reveal>
   178→        </div>
   179→
   180→        <PopIn delay={0.18} className="mt-6 flex flex-wrap gap-2">
   181→          {stats.map((s) => (
   182→            <span key={s} className="inline-flex items-center gap-1.5 rounded-full px-3 py-1.5 text-[12px] font-bold text-ink" style={{ background: tint(main, 12) }}>
   183→              <Sparkles className="h-3.5 w-3.5" style={{ color: main }} /> {s}
   184→            </span>
   185→          ))}
   186→        </PopIn>
   187→
   188→        <Squiggle color={main} className="mt-7 h-3.5 w-full opacity-60" />
   189→      </header>
   190→
   191→      {/* ————— contents ————— */}
   192→      <Reveal className="mx-auto max-w-3xl px-4 pt-4 sm:px-8">
   193→        <div className="rounded-[30px] bg-paper p-5 puff sm:p-6">
   194→          <div className="flex items-center gap-3">
   195→            <span className="grid h-10 w-10 shrink-0 place-items-center rounded-2xl" style={{ background: tint(main, 16) }}>
   196→              <Target className="h-5 w-5" style={{ color: shade(main, 62) }} />
   197→            </span>
   198→            <div>
   199→              <h2 className="font-display text-[18px] font-semibold text-ink">Contents</h2>
   200→              <p className="text-[12.5px] font-bold text-ink-faint">{doc.sections.length} sections · tap one to jump</p>
   201→            </div>
   202→          </div>
   203→          <div className="mt-4 grid grid-cols-2 gap-2 sm:grid-cols-3">
   204→            {doc.sections.map((s, i) => {
   205→              const accent = accentOf(i);
   206→              return (
   207→                <button
   208→                  key={s.spec}
   209→                  type="button"
   210→                  onClick={() => scrollToSpec(s.spec)}
   211→                  className={`group flex cursor-pointer items-center gap-2 rounded-[18px] px-3 py-2.5 text-left transition-transform hover:-translate-y-0.5${
   212→                    isDarkBand(i) ? " band-dark" : ""
   213→                  }`}
   214→                  style={isDarkBand(i) ? undefined : { background: bandOf(i) }}
   215→                >
   216→                  <span
   217→                    className="grid h-6 w-6 shrink-0 place-items-center rounded-full font-display text-[10.5px] font-bold"
   218→                    style={{ background: accent, color: readableInk(accent) }}
   219→                  >
   220→                    {s.spec}
   221→                  </span>
   222→                  <span className="truncate text-[12px] font-bold text-ink">{s.title}</span>
   223→                </button>
   224→              );
   225→            })}
   226→          </div>
   227→        </div>
   228→      </Reveal>
   229→
   230→      {/* ————— the string of rooms ————— */}
   231→      <div className="mt-10">
   232→        {/* sticky rail rides above every band */}
   233→        <div className="sticky top-[76px] z-40 mx-auto max-w-3xl px-4 sm:px-8">
   234→          <div
   235→            className="rounded-[24px] bg-paper/95 px-3.5 py-2.5 backdrop-blur-md"
   236→            style={{ boxShadow: "0 3px 0 var(--edge), 0 16px 34px -20px rgb(0 0 0 / 0.4)" }}
   237→          >
   238→            <div className="flex items-center gap-3">
   239→              {activeSection ? (
   240→                <span className="min-w-0 flex-1 truncate font-display text-[13.5px] font-semibold text-ink">
   241→                  <span style={{ color: shade(accentOf(activeIndex), 45) }}>{activeSection.spec}</span> · {activeSection.title}
   242→                </span>
   243→              ) : (
   244→                <span className="min-w-0 flex-1 truncate font-display text-[13.5px] font-semibold text-ink">{doc.title}</span>
   245→              )}
   246→            </div>
   247→            <div className="no-scrollbar mt-2 flex items-center gap-1.5 overflow-x-auto">
   248→              {doc.sections.map((s, i) => {
   249→                const isActive = active === s.spec;
   250→                const accent = accentOf(i);
   251→                return (
   252→                  <motion.button
   253→                    key={s.spec}
   254→                    type="button"
   255→                    whileTap={{ scale: 0.88 }}
   256→                    onClick={() => scrollToSpec(s.spec)}
   257→                    className="grid h-8 min-w-8 shrink-0 cursor-pointer place-items-center rounded-full px-2 font-display text-[11px] font-bold"
   258→                    style={
   259→                      isActive
   260→                        ? { background: accent, color: readableInk(accent), boxShadow: `0 3px 0 ${shade(accent)}` }
   261→                        : { background: tint(accent, 22), color: shade(accent, 62) }
   262→                    }
   263→                    title={`${s.spec} — ${s.title}`}
   264→                    aria-label={`Jump to spec ${s.spec}`}
   265→                    aria-current={isActive ? "true" : undefined}
   266→                  >
   267→                    {s.spec}
   268→                  </motion.button>
   269→                );
   270→              })}
   271→            </div>
   272→          </div>
   273→        </div>
   274→
   275→        <div className="mt-8">
   276→          <BandWave from="var(--color-cream)" to={bgOf(0)} />
   277→          {doc.sections.map((s, i) => {
   278→            const accent = accentOf(i);
   279→            const dark = isDarkBand(i);
   280→            const last = i === doc.sections.length - 1;
   281→            return (
   282→              <Fragment key={s.spec}>
   283→                <section
   284→                  id={`spec-${s.spec.replace(".", "-")}`}
   285→                  className={`relative scroll-mt-[190px]${dark ? " band-dark" : ""}`}
   286→                  style={dark ? undefined : { background: bandOf(i) }}
   287→                >
   288→                  <div className="mx-auto max-w-3xl px-4 py-14 sm:px-8 sm:py-16">
   289→                    <SpecSectionView section={s} accent={accent} />
   290→                  </div>
   291→                </section>
   292→                <BandWave from={bgOf(i)} to={last ? "var(--color-cream)" : bgOf(i + 1)} />
   293→              </Fragment>
   294→            );
   295→          })}
   296→        </div>
   297→      </div>
   298→
   299→      {/* ————— end of the note ————— */}
   300→      <div className="mx-auto max-w-3xl px-4 pt-10 pb-24 text-center sm:px-8">
   301→        <Reveal className="flex flex-col items-center gap-5">
   302→          <Squiggle color={main} className="h-3 w-40 opacity-70" />
   303→          <p className="font-display text-[19px] font-semibold text-ink">
   304→            That&apos;s the whole note — every spec point, word for word.
   305→          </p>
   306→          <SoftButton onClick={onBack} ariaLabel="Back to the topic page">
   307→            <ArrowLeft className="h-4.5 w-4.5" /> Back to the topic page
   308→          </SoftButton>
   309→          <p className="mt-2 text-[12px] leading-relaxed font-bold text-ink-faint">
   310→            <Fmt text={`==${unit.code}== · ${unit.title} · text kept verbatim from the ${doc.source}`} />
   311→          </p>
   312→        </Reveal>
   313→      </div>
   314→    </div>
   315→  );
   316→}

   358→/* ————— block dispatcher ————— */

   360→function BlockView({ block: b, accent }: { block: NotesBlock; accent: string }) {
   361→  switch (b.k) {
   362→    case "p":
   363→      return (
   364→        <p className="max-w-[68ch] text-[16.5px] leading-[1.8] text-ink/90">
   365→          <Fmt text={b.text} />
   366→        </p>
   367→      );
   368→    case "h":
   369→      return (
   370→        <div className="flex items-center gap-2.5 pt-2">
   371→          <span className="h-2.5 w-2.5 shrink-0 rounded-full" style={{ background: accent }} />
   372→          <h3 className="font-display text-[21px] font-semibold tracking-tight text-ink">
   373→            <Fmt text={b.text} />
   374→          </h3>
   375→        </div>
   376→      );
   377→    case "bullets":
   378→      return (
   379→        <ul className="flex max-w-[70ch] flex-col gap-2.5">
   380→          {b.items.map((t, i) => (
   381→            <li key={i} className="flex items-start gap-3 text-[16px] leading-[1.8] text-ink/90">
   382→              <span className="mt-[11px] h-2 w-2 shrink-0 rounded-full" style={{ background: tint(accent, 55) }} />
   383→              <span>
   384→                <Fmt text={t} />
   385→              </span>
   386→            </li>
   387→          ))}
   388→        </ul>
   389→      );
   390→    case "note":
   391→      return (
   392→        <div className="max-w-[72ch] rounded-[20px] px-4 py-3.5" style={{ background: tint("#FFD43B", 18) }}>
   393→          <div className="flex items-start gap-3">
   394→            <StickyNote className="mt-0.5 h-4.5 w-4.5 shrink-0 text-[#B08D00]" />
   395→            <p className="text-[14.5px] leading-[1.7] text-ink">
   396→              <span className="mr-1.5 font-display text-[11.5px] font-extrabold tracking-[0.12em] text-[#B08D00] uppercase">{b.label}:</span>
   397→              <Fmt text={b.text} />
   398→            </p>
   399→          </div>
   400→        </div>
   401→      );
   402→    case "def":
   403→      return (
   404→        <div className="max-w-[72ch] rounded-[20px] bg-paper px-5 py-4 puff" style={{ borderLeft: `5px solid ${tint(accent, 40)}` }}>
   405→          <h4 className="font-display text-[16.5px] font-semibold text-ink">
   406→            <Fmt text={b.term} />
   407→          </h4>
   408→          <p className="mt-1.5 text-[15.5px] leading-[1.75] text-ink-soft">
   409→            <Fmt text={b.text} />
   410→          </p>
   411→        </div>
   412→      );
   413→    case "eq":
   414→      return (
   415→        <div className="max-w-[72ch] rounded-[22px] p-5" style={{ background: tint(accent, 9) }}>
   416→          <div className="flex flex-wrap items-center gap-x-3 gap-y-2.5">
   417→            {b.eq.split(/\s{2,}/).map((part, i) => (
   418→              <span key={i} className="rounded-2xl bg-paper px-4 py-2.5 font-mono text-[14.5px] font-semibold text-ink shadow-[0_3px_0_color-mix(in_oklab,var(--color-ink)_7%,transparent)]">
   419→                <Fmt text={part.trim()} />
   420→              </span>
   421→            ))}
   422→          </div>
   423→          {b.note && (
   424→            <p className="mt-3 text-[13px] leading-relaxed font-bold text-ink-soft">
   425→              <Fmt text={b.note} />
   426→            </p>
   427→          )}
   428→        </div>
   429→      );
   430→    case "table":
   431→      return <CandyTable head={b.head} rows={b.rows} caption={b.caption} accent={accent} />;
   432→    case "worked":
   433→      return <Worked title={b.title} question={b.question} steps={b.steps} answer={b.answer} accent={accent} />;
   434→    case "steps":
   435→      return <Journey caption={b.caption} items={b.items} accent={accent} />;
   436→    case "checkpoint":
   437→      return (
   438→        <div className="flex items-center gap-3 rounded-full bg-paper px-5 py-4 puff">
   439→          <span className="grid h-10 w-10 shrink-0 place-items-center rounded-full" style={{ background: tint("#12B886", 18) }}>
   440→            <BookOpen className="h-5 w-5 text-mint" />
   441→          </span>
   442→          <p className="text-[15px] font-bold text-ink">
   443→            Checkpoint — the check point questions for this section are on page {b.page} of the textbook.
   444→          </p>
   445→        </div>
   446→      );
   447→    case "figure":
   448→      return <Figure kind={b.kind} caption={b.caption} accent={accent} />;
   449→    case "wow":
   450→      return (
   451→        <div className="max-w-[72ch] rounded-[22px] p-5 sm:p-6" style={{ background: tint("#9775FA", 12) }}>
   452→          <div className="flex items-start gap-3">
   453→            <span className="grid h-9 w-9 shrink-0 place-items-center rounded-2xl bg-paper">
   454→              <Lightbulb className="h-5 w-5 text-grape" />
   455→            </span>
   456→            <div>
   457→              <div className="font-display text-[11px] font-extrabold tracking-[0.18em] text-grape uppercase">Key idea</div>
   458→              <p className="mt-1 text-[15px] leading-[1.75] text-ink">
   459→                <Fmt text={b.text} />
   460→              </p>
   461→            </div>
   462→          </div>
   463→        </div>
   464→      );
   465→    default:
   466→      return null;
   467→  }
   468→}

   470→/* ————— candy table ————— */
   471→
   472→function CandyTable({
   473→  head,
   474→  rows,
   475→  caption,
   476→  accent,
