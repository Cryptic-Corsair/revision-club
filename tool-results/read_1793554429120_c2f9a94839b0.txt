    18→function ChartFrame({
    19→  w,

    21→  xLabel,

    53→        </text>
    54→      )}

    56→    </svg>
    57→  );

   163→/* ————— spectra ————— */
   164→
   165→interface Peak {
   166→  x: number;
   167→  h: number;
   168→  label: string;
   169→  sub?: string;
   170→  color?: string;
   171→}
   172→
   173→function Spectrum({
   174→  w = 560,
   175→  h = 250,
   176→  peaks,
   177→  xTicks,
   178→  xMax,
   179→  xMin = 0,
   180→  xLabel = "m/z",
   181→  yMax = 110,
   182→}: {
   183→  peaks: Peak[];
   184→  xTicks: number[];
   185→  xMax: number;
   186→  xMin?: number;
   187→  xLabel?: string;
   188→  yMax?: number;
   189→  w?: number;
   190→  h?: number;
   191→}) {
   192→  const pad = { l: 56, r: 20, t: 22, b: 42 };
   193→  const plotW = w - pad.l - pad.r;
   194→  const plotH = h - pad.t - pad.b;
   195→  const X = (v: number) => pad.l + ((v - xMin) / (xMax - xMin)) * plotW;
   196→  const Y = (v: number) => h - pad.b - (v / yMax) * plotH;
   197→  return (
   198→    <ChartFrame w={w} h={h} xLabel={xLabel} yLabel="relative abundance">
   199→      {[0, 50, 100].map((t) => (
   200→        <g key={t}>
   201→          <line x1={pad.l - 6} y1={Y(t)} x2={pad.l} y2={Y(t)} stroke={INK} strokeWidth="2" />
   202→          <text x={pad.l - 10} y={Y(t) + 4} textAnchor="end" fontSize="11" fontWeight="700" fill={INK} fontFamily="var(--font-body)">
   203→            {t}
   204→          </text>
   205→        </g>
   206→      ))}
   207→      {xTicks.map((t, i) => (
   208→        <text key={i} x={X(t)} y={h - pad.b + 18} textAnchor="middle" fontSize="11" fontWeight="700" fill={INK} fontFamily="var(--font-body)">
   209→          {t}
   210→        </text>
   211→      ))}
   212→      {peaks.map((p, i) => (
   213→        <g key={i}>
   214→          <line x1={X(p.x)} y1={Y(0)} x2={X(p.x)} y2={Y(p.h)} stroke={p.color ?? INK} strokeWidth="4.5" strokeLinecap="round" />
   215→          {p.label && (
   216→            <text x={X(p.x)} y={Y(p.h) - 8} textAnchor="middle" fontSize="11.5" fontWeight="800" fill={p.color ?? INK} fontFamily="var(--font-body)">
   217→              {p.label}
   218→            </text>
   219→          )}
   220→        </g>
   221→      ))}
   222→    </ChartFrame>
   223→  );
   224→}

   406→/* ————— 2.14 · box notation + filling order ————— */
   407→
   408→const CAP: Record<string, number> = { s: 2, p: 6, d: 10, f: 14 };
   409→const ORBITALS: Record<string, number> = { s: 1, p: 3, d: 5, f: 7 };
   410→const ORDER = ["1s", "2s", "2p", "3s", "3p", "4s", "3d", "4p"];
   411→
   412→export function buildConfig(z: number): { sub: string; e: number }[] {
   413→  const parts: { sub: string; e: number }[] = [];
   414→  let left = z;
   415→  for (const sub of ORDER) {
   416→    if (left <= 0) break;
   417→    const e = Math.min(CAP[sub[1]], left);
   418→    parts.push({ sub, e });
   419→    left -= e;
   420→  }
   421→  if (z === 24) {
   422→    // Cr: donate one 4s electron to 3d → half-filled d
   423→    for (const p of parts) {
   424→      if (p.sub === "4s") p.e = 1;
   425→      if (p.sub === "3d") p.e = 5;
   426→    }
   427→  }
   428→  if (z === 29) {
   429→    // Cu: donate one 4s electron to 3d → filled d
   430→    for (const p of parts) {
   431→      if (p.sub === "4s") p.e = 1;
   432→      if (p.sub === "3d") p.e = 10;
   433→    }
   434→  }
   435→  return parts;
   436→}

   438→export function spdfString(parts: { sub: string; e: number }[]): string {
   439→  return parts.map((p) => `${p.sub}${p.e}`).join(" ");
   440→}
   441→
   442→export function BoxNotation({
   443→  parts,
   444→  main = "#FF6B6B",
   445→  compact = false,
   446→}: {
   447→  parts: { sub: string; e: number }[];
   448→  main?: string;
   449→  compact?: boolean;
   450→}) {
   451→  const boxSize = compact ? 20 : 30;
   452→  return (
   453→    <div className="flex flex-wrap items-end gap-x-4 gap-y-3">
   454→      {parts.map((p) => {
   455→        const orbitals = ORBITALS[p.sub[1]] ?? 1;
   456→        const boxes = Array.from({ length: orbitals }, (_, i) => {
   457→          const up = i < p.e;
   458→          const down = i < p.e - orbitals;
   459→          return { up, down };
   460→        });
   461→        return (
   462→          <div key={p.sub} className="flex flex-col items-center gap-1.5">
   463→            <div className="flex gap-1">
   464→              {boxes.map((b, i) => (
   465→                <div
   466→                  key={i}
   467→                  className="grid place-items-center rounded-[8px] bg-white font-bold"
   468→                  style={{
   469→                    width: boxSize,
   470→                    height: boxSize,
   471→                    boxShadow: "inset 0 0 0 2.5px rgba(61,43,31,0.10), 0 2.5px 0 rgba(61,43,31,0.06)",
   472→                    fontSize: compact ? 12 : 15,
   473→                  }}
   474→                >
   475→                  {b.up && (
   476→                    <span style={{ color: shade(main) }}>
   477→                      ↑{b.down ? <span style={{ color: "#12B886" }}>↓</span> : null}
   478→                    </span>
   479→                  )}
   480→                </div>
   481→              ))}
   482→            </div>
   483→            <span className="font-mono text-[11.5px] font-bold text-[#8A7261]">
   484→              {p.sub}
   485→              {p.e}
   486→            </span>
   487→          </div>
   488→        );
   489→      })}
   490→    </div>
   491→  );
   492→}
