    46→/* ═══════════════════════════════════════════════════════════════
    47→   Reaction Club · notes reader — one enclosed room per spec point.
    48→   Each section is a full-width tonal band borrowing one candy
    49→   colour, walking the hue wheel so neighbours always contrast.
    50→   In the hybrid lab every third band is a deep cocoa room with
    51→   milk-toned fonts (see html.hybrid .room in globals.css); in the
    52→   other two labs the same bands stay candy — the class is inert.
    53→   The footer's wave rides between the bands, so the note reads as
    54→   a string of connected rooms rather than one long scroll.
    55→   ═══════════════════════════════════════════════════════════════ */
    56→
    57→/* the rotating palette — hue-wheel order: red → orange → yellow →
    58→   green → purple → pink, so adjacent bands always differ */
    59→const BANDS = ["#FF6B6B", "#FF922B", "#FFD43B", "#12B886", "#9775FA", "#F06595"];
    60→
    61→const WAVE_D = "M0,52 C160,92 340,8 540,30 C740,52 900,96 1100,68 C1260,46 1370,22 1440,40 L1440,90 L0,90 Z";
    62→
    63→function BandWave({ from, to }: { from: string; to: string }) {
    64→  return (
    65→    <div aria-hidden style={{ background: from }}>
    66→      <svg viewBox="0 0 1440 90" preserveAspectRatio="none" className="block h-[46px] w-full sm:h-[68px]">
    67→        {/* fill rides on style so var() strings with fallbacks resolve */}
    68→        <path d={WAVE_D} style={{ fill: to }} />
    69→      </svg>
    70→    </div>
    71→  );
    72→}

    84→/* ————— main reader ————— */
    85→
    86→export function NotesReader({ doc, unit, onBack }: { doc: NotesDoc; unit: Unit; onBack: () => void }) {
    87→  const main = unit.candy.main;
    88→  const [active, setActive] = useState<string>(doc.sections[0]?.spec ?? "");
    89→
    90→  const { scrollYProgress } = useScroll();
    91→  const scaleX = useSpring(scrollYProgress, { stiffness: 140, damping: 26, mass: 0.4 });
    92→
    93→  /* scroll spy */
    94→  const spy = useRef<IntersectionObserver | null>(null);
    95→  useEffect(() => {
    96→    spy.current = new IntersectionObserver(
    97→      (entries) => {
    98→        for (const e of entries) {
    99→          if (e.isIntersecting) setActive(e.target.id.replace("spec-", "").replace("-", "."));
   100→        }
   101→      },
   102→      { rootMargin: "-30% 0px -60% 0px" }
   103→    );
   104→    for (const s of doc.sections) {
   105→      const el = document.getElementById(`spec-${s.spec.replace(".", "-")}`);
   106→      if (el) spy.current?.observe(el);
   107→    }
   108→    return () => spy.current?.disconnect();
   109→  }, [doc]);
   110→
   111→  const count = (k: NotesBlock["k"]) => {
   112→    let n = 0;
   113→    const walk = (blocks: NotesBlock[]) => blocks.forEach((b) => b.k === k && n++);
   114→    walk(doc.intro);
   115→    doc.sections.forEach((s) => walk(s.blocks));
   116→    if (doc.outro) walk(doc.outro);
   117→    return n;
   118→  };
   119→  const stats = [
   120→    `${doc.sections.length} spec sections`,
   121→    `${count("worked")} worked examples`,
   122→    `${count("figure")} diagrams`,
   123→    `${count("checkpoint")} textbook checkpoints`,
   124→  ];
   125→
   126→  const accentOf = (i: number) => BANDS[i % BANDS.length];
   127→  /* bands mix their candy over --tint-base at --band-mix — 10% by
   128→     day, 14% after dark, so roast-tone neighbours stay distinct */
   129→  const bandOf = (i: number) =>
   130→    `color-mix(in oklab, ${accentOf(i)} var(--band-mix, 10%), var(--tint-base))`;
   131→  /* every third room goes cocoa in the hybrid lab (2.2, 2.5, 2.8 …).
   132→     Elsewhere --room-surface is undefined and the fallback keeps the
   133→     plain candy band — one expression, correct in all three labs. */
   134→  const isRoom = (i: number) => i % 3 === 1;
   135→  const bgOf = (i: number) => (isRoom(i) ? `var(--room-surface, ${bandOf(i)})` : bandOf(i));
   136→  const activeIndex = Math.max(0, doc.sections.findIndex((s) => s.spec === active));
   137→  const activeSection = doc.sections[activeIndex];
   138→
   139→  return (
   140→    <div className="relative">
   141→      {/* reading progress */}
   142→      <motion.div
   143→        aria-hidden
   144→        className="fixed top-0 left-0 z-[70] h-[7px] w-full origin-left rounded-r-full"
   145→        style={{ scaleX, background: `linear-gradient(90deg, ${main}, #FFD43B)` }}
   146→      />
   147→
   148→      {/* ————— header ————— */}
   149→      <header className="mx-auto max-w-3xl px-4 pt-24 sm:px-8">
   150→        <PopIn>
   151→          <SoftButton onClick={onBack} className="min-h-[42px] px-4 py-2 text-[13.5px]" ariaLabel="Back to the topic page">
   152→            <ArrowLeft className="h-4 w-4" /> Back to the topic page
   153→          </SoftButton>
   154→        </PopIn>
   155→
   156→        <PopIn delay={0.05} className="mt-7 flex flex-wrap items-center gap-2">
   157→          <span
   158→            className="inline-flex -rotate-1 items-center gap-1.5 rounded-full px-3.5 py-1.5 font-display text-[11.5px] font-bold tracking-[0.14em] text-white uppercase"
   159→            style={{ background: main }}
   160→          >
   161→            <BookOpen className="h-3.5 w-3.5" /> Revision note
   162→          </span>
   163→          <span className="inline-flex items-center rounded-full bg-ink/5 px-3.5 py-1.5 text-[12.5px] font-bold text-ink-soft">
   164→            {doc.source}
   165→          </span>
   166→        </PopIn>
   167→        <PopIn delay={0.1}>
   168→          <p className="mt-6 font-display text-[15px] font-semibold tracking-[0.06em] uppercase" style={{ color: inkMix(main, 52) }}>
   169→            {doc.kicker}
   170→          </p>
   171→          <h1 className="mt-2 font-display text-[clamp(2.1rem,5vw,3.2rem)] leading-[1.05] font-semibold tracking-tight text-ink">
   172→            {doc.title}
   173→          </h1>
   174→        </PopIn>
   175→
   176→        <div className="mt-6 flex flex-col items-start gap-5 sm:flex-row sm:items-center">
   177→          <div className="shrink-0">
   178→            <Mascot main={main} size={104} bubbles={false} />
   179→          </div>
   180→          <Reveal delay={0.15} className="min-w-0">
   181→            <SpeechBubble tailClass="left-8 -top-[7px] rotate-[225deg]">
   182→              <p className="text-[14.5px] leading-[1.75] text-ink">
   183→                {doc.intro.map((b, i) =>
   184→                  b.k === "p" ? (
   185→                    <span key={i} className="block">
   186→                      <Fmt text={b.text} />
   187→                    </span>
   188→                  ) : null
   189→                )}
   190→              </p>
   191→            </SpeechBubble>
   192→          </Reveal>
   193→        </div>
   194→
   195→        <PopIn delay={0.18} className="mt-6 flex flex-wrap gap-2">
   196→          {stats.map((s) => (
   197→            <span key={s} className="inline-flex items-center gap-1.5 rounded-full px-3 py-1.5 text-[12px] font-bold text-ink" style={{ background: tint(main, 12) }}>
   198→              <Sparkles className="h-3.5 w-3.5" style={{ color: main }} /> {s}
   199→            </span>
   200→          ))}
   201→        </PopIn>
   202→
   203→        <Squiggle color={main} className="mt-7 h-3.5 w-full opacity-60" />
   204→      </header>
   205→
   206→      {/* ————— contents ————— */}
   207→      <Reveal className="mx-auto max-w-3xl px-4 pt-4 sm:px-8">
   208→        <div className="rounded-[30px] bg-paper p-5 puff sm:p-6">
   209→          <div className="flex items-center gap-3">
   210→            <span className="grid h-10 w-10 shrink-0 place-items-center rounded-2xl" style={{ background: tint(main, 16) }}>
   211→              <Target className="h-5 w-5" style={{ color: inkMix(main, 60) }} />
   212→            </span>
   213→            <div>
   214→              <h2 className="font-display text-[18px] font-semibold text-ink">Contents</h2>
   215→              <p className="text-[12.5px] font-bold text-ink-faint">{doc.sections.length} sections · tap one to jump</p>
   216→            </div>
   217→          </div>
   218→          <div className="mt-4 grid grid-cols-2 gap-2 sm:grid-cols-3">
   219→            {doc.sections.map((s, i) => {
   220→              const accent = accentOf(i);
   221→              return (
   222→                <button
   223→                  key={s.spec}
   224→                  type="button"
   225→                  onClick={() => scrollToSpec(s.spec)}
   226→                  className={`group flex cursor-pointer items-center gap-2 rounded-[18px] px-3 py-2.5 text-left transition-transform hover:-translate-y-0.5${
   227→                    isRoom(i) ? " room" : ""
   228→                  }`}
   229→                  style={{ background: bgOf(i) }}
   230→                >
   231→                  <span
   232→                    className="grid h-6 w-6 shrink-0 place-items-center rounded-full font-display text-[10.5px] font-bold"
   233→                    style={{ background: accent, color: readableInk(accent) }}
   234→                  >
   235→                    {s.spec}
   236→                  </span>
   237→                  <span className="truncate text-[12px] font-bold text-ink">{s.title}</span>
   238→                </button>
   239→              );
   240→            })}
   241→          </div>
   242→        </div>
   243→      </Reveal>
   244→
   245→      {/* ————— the string of rooms ————— */}
   246→      <div className="mt-10">
   247→        {/* sticky rail rides above every band */}
   248→        <div className="sticky top-[76px] z-40 mx-auto max-w-3xl px-4 sm:px-8">
   249→          <div
   250→            className="rounded-[24px] bg-paper/95 px-3.5 py-2.5 backdrop-blur-md"
   251→            style={{ boxShadow: "0 3px 0 var(--edge), 0 16px 34px -20px rgb(0 0 0 / 0.4)" }}
   252→          >
   253→            <div className="flex items-center gap-3">
   254→              {activeSection ? (
   255→                <span className="min-w-0 flex-1 truncate font-display text-[13.5px] font-semibold text-ink">
   256→                  <span style={{ color: inkMix(accentOf(activeIndex), 45) }}>{activeSection.spec}</span> · {activeSection.title}
   257→                </span>
   258→              ) : (
   259→                <span className="min-w-0 flex-1 truncate font-display text-[13.5px] font-semibold text-ink">{doc.title}</span>
   260→              )}
   261→            </div>
   262→            <div className="no-scrollbar mt-2 flex items-center gap-1.5 overflow-x-auto">
   263→              {doc.sections.map((s, i) => {
   264→                const isActive = active === s.spec;
   265→                const accent = accentOf(i);
   266→                return (
   267→                  <motion.button
   268→                    key={s.spec}
   269→                    type="button"
   270→                    whileTap={{ scale: 0.88 }}
   271→                    onClick={() => scrollToSpec(s.spec)}
   272→                    className="grid h-8 min-w-8 shrink-0 cursor-pointer place-items-center rounded-full px-2 font-display text-[11px] font-bold"
   273→                    style={
   274→                      isActive
   275→                        ? { background: accent, color: readableInk(accent), boxShadow: `0 3px 0 ${shade(accent)}` }
   276→                        : { background: tint(accent, 22), color: inkMix(accent, 62) }
   277→                    }
   278→                    title={`${s.spec} — ${s.title}`}
   279→                    aria-label={`Jump to spec ${s.spec}`}
   280→                    aria-current={isActive ? "true" : undefined}
   281→                  >
   282→                    {s.spec}
   283→                  </motion.button>
   284→                );
   285→              })}
   286→            </div>
   287→          </div>
   288→        </div>
   289→
   290→        <div className="mt-8">
   291→          <BandWave from="var(--color-cream)" to={bgOf(0)} />
   292→          {doc.sections.map((s, i) => {
   293→            const accent = accentOf(i);
   294→            const room = isRoom(i);
   295→            const last = i === doc.sections.length - 1;
   296→            return (
   297→              <Fragment key={s.spec}>
   298→                <section
   299→                  id={`spec-${s.spec.replace(".", "-")}`}
   300→                  className={`relative scroll-mt-[190px]${room ? " room" : ""}`}
   301→                  style={{ background: bgOf(i) }}
   302→                >
   303→                  <div className="mx-auto max-w-3xl px-4 py-14 sm:px-8 sm:py-16">
   304→                    <SpecSectionView section={s} accent={accent} />
   305→                  </div>
   306→                </section>
   307→                <BandWave from={bgOf(i)} to={last ? "var(--color-cream)" : bgOf(i + 1)} />
   308→              </Fragment>
   309→            );
   310→          })}
   311→        </div>
   312→      </div>
   313→
   314→      {/* ————— end of the note ————— */}
   315→      <div className="mx-auto max-w-3xl px-4 pt-10 pb-24 text-center sm:px-8">
   316→        <Reveal className="flex flex-col items-center gap-5">
   317→          <Squiggle color={main} className="h-3 w-40 opacity-70" />
   318→          <p className="font-display text-[19px] font-semibold text-ink">
   319→            That&apos;s the whole note — every spec point, word for word.
   320→          </p>
   321→          <SoftButton onClick={onBack} ariaLabel="Back to the topic page">
   322→            <ArrowLeft className="h-4.5 w-4.5" /> Back to the topic page
   323→          </SoftButton>
   324→          <p className="mt-2 text-[12px] leading-relaxed font-bold text-ink-faint">
   325→            <Fmt text={`==${unit.code}== · ${unit.title} · text kept verbatim from the ${doc.source}`} />
   326→          </p>
   327→        </Reveal>
   328→      </div>
   329→    </div>
   330→  );
   331→}

   388→            <Fmt text={b.text} />
   389→          </h3>
   390→        </div>
   391→      );

   412→              <Fmt text={b.text} />

   420→          <h4 className="font-display text-[16.5px] font-semibold text-ink">

   431→          <div className="flex flex-wrap items-center gap-x-3 gap-y-2.5">
   432→            {b.eq.split(/\s{2,}/).map((part, i) => (
   433→              <span key={i} className="rounded-2xl bg-paper px-4 py-2.5 font-mono text-[14.5px] font-semibold text-ink shadow-[0_3px_0_color-mix(in_oklab,var(--color-ink)_7%,transparent)]">
   434→                <Fmt text={part.trim()} />

   440→              <Fmt text={b.note} />

   453→        <div className="flex items-center gap-3 rounded-full bg-paper px-5 py-4 puff">

   455→            <BookOpen className="h-5 w-5 text-mint" />

   460→        </div>

   463→      return <Figure kind={b.kind} caption={b.caption} accent={accent} />;
   464→    case "wow":
   465→      return (

   663→function Figure({ kind, caption, accent }: { kind: string; caption?: string; accent: string }) {

   667→    case "notation": node = <FigNotation accent={accent} />; break;
   668→    case "mass-spec": node = <FigMassSpec />; break;
   669→    case "shielding": node = <FigShielding accent={accent} />; break;
   670→    case "spectrum-neon": node = <FigSpectrumNeon />; break;
   671→    case "spectrum-boron": node = <FigSpectrumBoron />; break;
   672→    case "spectrum-bromine": node = <FigSpectrumBromine />; break;
   673→    case "spectrum-cl2": node = <FigSpectrumCl2 />; break;
   674→    case "spectrum-ethanol": node = <FigSpectrumEthanol />; break;
   675→    case "orbitals": node = <FigOrbitals accent={accent} />; break;
   676→    case "fill-demo": node = <FigFillDemo accent={accent} />; break;
   677→    case "fill-order": node = <FigFillOrder accent={accent} />; break;
   678→    case "boxes-c": node = <FigBoxesC accent={accent} />; break;
