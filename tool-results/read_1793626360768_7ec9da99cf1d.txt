     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import { AnimatePresence, motion } from "framer-motion";
     5→import {
     6→  ArrowLeft,
     7→  Check,
     8→  Flame,
     9→  RotateCcw,
    10→  Sparkles,
    11→  Trophy,
    12→  X,
    13→  Zap,
    14→} from "./icons";
    15→import type { UnitCode } from "@/lib/spectra/types";
    16→import { DECK, DECK_BY_UNIT, shuffle, type Card } from "@/lib/spectra/deck";
    17→import { UNITS } from "@/lib/spectra/units";
    18→import { recordRun, useBlendStats } from "@/lib/spectra/blend-stats";
    19→import { useRevised } from "@/lib/spectra/progress";
    20→import { Fmt } from "./format";
    21→import {
    22→  Blob,
    23→  Chip,
    24→  Confetti,
    25→  DoodleSparkle,
    26→  Float,
    27→  Mascot,
    28→  PopIn,
    29→  Reveal,
    30→  SectionTag,
    31→  SoftButton,
    32→  SpeechBubble,
    33→  tint,
    34→  shade,
    35→} from "./bits";
    36→
    37→/* ═══════════════════════════════════════════════════════════════
    38→   Reaction Club · BLEND MODE
    39→   Every checkpoint question on the site, thrown into a blender.
    40→   Flip → self-judge → streaks grow → confetti at combos.
    41→   Keyboard: Space flip · → got it · ← not yet
    42→   ═══════════════════════════════════════════════════════════════ */
    43→
    44→type Phase = "idle" | "playing" | "done";
    45→
    46→const MILESTONES = [3, 5, 8, 12, 16, 20];
    47→
    48→const COMBO_QUIPS: Record<number, string> = {
    49→  3: "Three in a row.",
    50→  5: "Five in a row — keep going.",
    51→  8: "Eight in a row.",
    52→  12: "Twelve in a row.",
    53→  16: "Sixteen in a row.",
    54→  20: "Twenty in a row. Excellent.",
    55→};
    56→
    57→export function Blend({
    58→  onBack,
    59→  onOpenTopic,
    60→}: {
    61→  onBack: () => void;
    62→  onOpenTopic: (code: UnitCode, topicId: string) => void;
    63→}) {
    64→  const stats = useBlendStats();
    65→  const revised = useRevised();
    66→
    67→  const [phase, setPhase] = useState<Phase>("idle");
    68→  const [filter, setFilter] = useState<UnitCode | "all">("all");
    69→  const [run, setRun] = useState<{ cards: Card[]; picks: ("hit" | "miss")[] } | null>(null);
    70→
    71→  const pile = useMemo(
    72→    () => (filter === "all" ? DECK : DECK.filter((c) => c.unit === filter)),
    73→    [filter]
    74→  );
    75→
    76→  const start = (size: number) => {
    77→    const picked = shuffle(pile).slice(0, size);
    78→    setFiredMilestone(0);
    79→    setRun({ cards: picked, picks: [] });
    80→    setPhase("playing");
    81→    window.setTimeout(
    82→      () => stageRef.current?.scrollIntoView({ behavior: "smooth", block: "start" }),
    83→      80
    84→    );
    85→  };
    86→
    87→  const [firedMilestone, setFiredMilestone] = useState(0);
    88→  const [confetti, setConfetti] = useState(0);
    89→  const [quip, setQuip] = useState<string | null>(null);
    90→  const quipTimer = useRef<number | undefined>(undefined);
    91→
    92→  /* ————— derived run state ————— */
    93→  const idx = run ? run.picks.length : 0;
    94→  const card = run && idx < run.cards.length ? run.cards[idx] : null;
    95→  const hits = run ? run.picks.filter((p) => p === "hit").length : 0;
    96→
    97→  let streak = 0;
    98→  let bestStreak = 0;
    99→  for (const p of run?.picks ?? []) {
   100→    if (p === "hit") {
   101→      streak++;
   102→      bestStreak = Math.max(bestStreak, streak);
   103→    } else streak = 0;
   104→  }
   105→
   106→  const judge = (verdict: "hit" | "miss") => {
   107→    if (!run || run.picks.length >= run.cards.length) return;
   108→    const picks = [...run.picks, verdict];
   109→    setRun({ ...run, picks });
   110→
   111→    // streak bookkeeping (event-driven — no effects, no cascades)
   112→    let s = 0;
   113→    let bs = 0;
   114→    for (const p of picks) {
   115→      if (p === "hit") bs = Math.max(bs, ++s);
   116→      else s = 0;
   117→    }
   118→    if (bs >= 3) {
   119→      const m = MILESTONES.reduce((acc, x) => (bs >= x ? x : acc), 0);
   120→      if (m > firedMilestone) {
   121→        setFiredMilestone(m);
   122→        setConfetti((c) => c + 1);
   123→        setQuip(COMBO_QUIPS[m] ?? "Unstoppable.");
   124→        window.clearTimeout(quipTimer.current);
   125→        quipTimer.current = window.setTimeout(() => setQuip(null), 3200);
   126→      }
   127→    }
   128→
   129→    // last card → record the run and show results
   130→    if (picks.length === run.cards.length) {
   131→      const h = picks.filter((p) => p === "hit").length;
   132→      recordRun(h, picks.length, bs);
   133→      if (picks.length > 0 && h / picks.length >= 0.7) setConfetti((c) => c + 1);
   134→      setPhase("done");
   135→    }
   136→  };
   137→
   138→  const flipRef = useRef<(() => void) | null>(null);
   139→
   140→  /* ————— keyboard ————— */
   141→  useEffect(() => {
   142→    if (phase !== "playing") return;
   143→    const onKey = (e: KeyboardEvent) => {
   144→      if (e.repeat) return;
   145→      if (e.code === "Space" || e.key === " ") {
   146→        e.preventDefault();
   147→        flipRef.current?.();
   148→      } else if (e.key === "ArrowRight") judge("hit");
   149→      else if (e.key === "ArrowLeft") judge("miss");
   150→    };
   151→    window.addEventListener("keydown", onKey);
   152→    return () => window.removeEventListener("keydown", onKey);
   153→  });
   154→
   155→  const stageRef = useRef<HTMLElement | null>(null);
   156→
   157→  return (
   158→    <div className="relative mx-auto max-w-3xl px-5 pt-24 pb-20 sm:px-8">
   159→      {/* back */}
   160→      <PopIn>
   161→        <SoftButton onClick={onBack} className="min-h-[42px] px-4 py-2 text-[13.5px]">
   162→          <ArrowLeft className="h-4 w-4" /> back to all units
   163→        </SoftButton>
   164→      </PopIn>
   165→
   166→      {/* ————— header ————— */}
   167→      <header className="mt-8">
   168→        <PopIn delay={0.05}>
   169→          <SectionTag main="#9775FA" tilt={-2}>
   170→            flashcards
   171→          </SectionTag>
   172→        </PopIn>
   173→        <PopIn delay={0.1}>
   174→          <h1 className="mt-5 font-display text-[clamp(2.3rem,5vw,3.4rem)] leading-[1.05] font-semibold tracking-tight text-ink">
   175→            Checkpoint questions,{" "}
   176→            <span className="relative inline-block -rotate-2 rounded-[0.35em] bg-grape/15 px-[0.18em]">
   177→              shuffled.
   178→            </span>
   179→          </h1>
   180→          <p className="mt-3.5 max-w-[58ch] text-[16.5px] leading-[1.8] text-ink-soft">
   181→            Every checkpoint question on the site, shuffled into one deck. Flip a card, answer
   182→            it, then reveal and compare. {" "}
   183→            <strong className="font-extrabold text-ink">Space</strong> flips,{" "}
   184→            <strong className="font-extrabold text-ink">← →</strong> mark it.
   185→          </p>
   186→        </PopIn>
   187→      </header>
   188→
   189→      {/* ————— scoreboard ————— */}
   190→      <PopIn delay={0.16} className="mt-8 flex flex-wrap gap-2.5">
   191→        <span className="inline-flex items-center gap-2 rounded-full bg-grape/15 px-4 py-2.5 text-[13.5px] font-extrabold text-ink">
   192→          <Trophy className="h-4 w-4 text-grape" /> best streak: {stats.best}
   193→        </span>
   194→        <span className="inline-flex items-center gap-2 rounded-full bg-mint/15 px-4 py-2.5 text-[13.5px] font-extrabold text-ink">
   195→          <Zap className="h-4 w-4 text-mint" /> {stats.blends} rounds played
   196→        </span>
   197→        {stats.tries > 0 && (
   198→          <span className="inline-flex items-center gap-2 rounded-full bg-lemon/20 px-4 py-2.5 text-[13.5px] font-extrabold text-ink">
   199→            <Sparkles className="h-4 w-4 text-lemon" />
   200→            {Math.round((stats.hits / stats.tries) * 100)}% all-time correct
   201→          </span>
   202→        )}
   203→      </PopIn>
   204→
   205→      {/* ————— IDLE: pick a pile ————— */}
   206→      {phase === "idle" && (
   207→        <section className="mt-10">
   208→          <Reveal>
   209→            <div className="mb-3 flex items-center gap-2 font-display text-[13px] font-bold tracking-[0.16em] text-ink-faint uppercase">
   210→              choose a deck <DoodleSparkle className="h-4 w-4 text-lemon" />
   211→            </div>
   212→            <div className="flex flex-wrap gap-2.5">
   213→              <motion.button
   214→                type="button"
   215→                whileHover={{ y: -2 }}
   216→                whileTap={{ scale: 0.95 }}
   217→                onClick={() => setFilter("all")}
   218→                className="cursor-pointer rounded-full px-4 py-2.5 font-display text-[13.5px] font-bold"
   219→                style={
   220→                  filter === "all"
   221→                    ? { background: "var(--color-ink)", color: "var(--color-cream)", boxShadow: "0 3px 0 color-mix(in oklab, var(--color-ink) 55%, #000)" }
   222→                    : { background: "color-mix(in oklab, var(--color-ink) 7%, transparent)", color: "var(--color-ink-soft)" }
   223→                }
   224→              >
   225→                the whole course · {DECK.length}
   226→              </motion.button>
   227→              {UNITS.map((u) => (
   228→                <motion.button
   229→                  key={u.code}
   230→                  type="button"
   231→                  whileHover={{ y: -2 }}
   232→                  whileTap={{ scale: 0.95 }}
   233→                  onClick={() => setFilter(u.code)}
   234→                  disabled={DECK_BY_UNIT[u.code] === 0}
   235→                  className="cursor-pointer rounded-full px-4 py-2.5 font-display text-[13.5px] font-bold disabled:cursor-not-allowed disabled:opacity-40"
   236→                  style={
   237→                    filter === u.code
   238→                      ? { background: u.candy.main, color: "#fff", boxShadow: `0 3px 0 ${shade(u.candy.main)}` }
   239→                      : { background: tint(u.candy.main, 14), color: shade(u.candy.main, 62) }
   240→                  }
   241→                >
   242→                  {u.code.replace("WCH", "")} · {DECK_BY_UNIT[u.code]} cards
   243→                </motion.button>
   244→              ))}
   245→            </div>
   246→          </Reveal>
   247→
   248→          <Reveal delay={0.1} className="mt-10">
   249→            <div
   250→              className="relative flex flex-col items-center gap-6 overflow-hidden rounded-[40px] p-7 text-center sm:p-10"
   251→              style={{ background: tint("#9775FA", 9) }}
   252→            >
   253→              <Blob color={tint("#9775FA", 20)} className="top-[-20%] right-[-10%] h-[60%] w-[45%]" />
   254→              <Float>
   255→                <Mascot main="#9775FA" mood="cheer" size={150} />
   256→              </Float>
   257→              <div className="relative z-10">
   258→                <h2 className="font-display text-[clamp(1.5rem,3.4vw,2.1rem)] font-semibold text-ink">
   259→                  {pile.length} cards in this deck
   260→                </h2>
   261→                <p className="mx-auto mt-2 max-w-[46ch] text-[14.5px] leading-relaxed text-ink-soft">
   262→                  Ten cards per round, shuffled each time. Answer first, then check the
   263→                  solution.
   264→                </p>
   265→                <motion.button
   266→                  type="button"
   267→                  onClick={() => start(Math.min(10, pile.length))}
   268→                  whileHover={{ y: -2 }}
   269→                  whileTap={{ scale: 0.95, y: 2 }}
   270→                  className="candy mt-6 inline-flex min-h-[56px] cursor-pointer items-center gap-2.5 rounded-full px-8 py-4 font-display text-[17px] font-semibold text-white select-none"
   271→                  style={{ "--candy-bg": "#9775FA", "--candy-side": shade("#9775FA") } as React.CSSProperties}
   272→                >
   273→                  <Zap className="h-5 w-5" /> Shuffle {Math.min(10, pile.length)} cards
   274→                </motion.button>
   275→                {pile.length > 10 && (
   276→                  <div className="mt-4 flex flex-wrap justify-center gap-2">
   277→                    {[15, 25].filter((n) => n <= pile.length).map((n) => (
   278→                      <motion.button
   279→                        key={n}
   280→                        type="button"
   281→                        onClick={() => start(n)}
   282→                        whileHover={{ y: -2 }}
   283→                        whileTap={{ scale: 0.95 }}
   284→                        className="cursor-pointer rounded-full bg-paper/80 px-4 py-2 text-[12.5px] font-extrabold text-ink-soft"
   285→                      >
   286→                        {n} cards
   287→                      </motion.button>
   288→                    ))}
   289→                  </div>
   290→                )}
   291→              </div>
   292→            </div>
   293→          </Reveal>
   294→        </section>
   295→      )}
   296→
   297→      {/* ————— PLAYING: the card ————— */}
   298→      {phase === "playing" && run && card && (
   299→        <section ref={stageRef} className="mt-10 scroll-mt-24">
   300→          {/* progress dots — colours ride on style (motion can't tween
   301→              color-mix strings), only the scale springs */}
   302→          <div className="mb-5 flex items-center justify-center gap-1.5">
   303→            {run.cards.map((_, i) => (
   304→              <motion.span
   305→                key={i}
   306→                aria-hidden
   307→                animate={{ scale: i === idx ? 1.5 : 1 }}
   308→                transition={{ type: "spring", stiffness: 400, damping: 18 }}
   309→                style={{
   310→                  backgroundColor:
   311→                    i < idx
   312→                      ? run.picks[i] === "hit"
   313→                        ? "#12b886"
   314→                        : "#ff6b6b"
   315→                      : i === idx
   316→                        ? "var(--color-ink)"
   317→                        : "color-mix(in oklab, var(--color-ink) 16%, transparent)",
   318→                }}
   319→                className="h-2.5 w-2.5 rounded-full"
   320→              />
   321→            ))}
   322→          </div>
   323→
   324→          {/* streak meter */}
   325→          <div className="mb-5 flex items-center justify-center gap-2">
   326→            <motion.span
   327→              key={streak}
   328→              initial={{ scale: 0.6, rotate: -8 }}
   329→              animate={{ scale: 1, rotate: 0 }}
   330→              transition={{ type: "spring", stiffness: 400, damping: 12 }}
   331→              className={`inline-flex items-center gap-1.5 rounded-full px-4 py-2 font-display text-[14px] font-bold ${
   332→                streak >= 3 ? "text-white" : "bg-ink/5 text-ink-soft"
   333→              }`}
   334→              style={streak >= 3 ? { background: "#ff6b6b", boxShadow: "0 3px 0 #e8590c" } : undefined}
   335→            >
   336→              <Flame className="h-4 w-4" strokeWidth={2.5} />
   337→              streak {streak}
   338→              {bestStreak >= 3 && streak === bestStreak && streak > 0 && (
   339→                <span className="ml-0.5 text-[11px] opacity-80">· best so far</span>
   340→              )}
   341→            </motion.span>
   342→            <span className="rounded-full bg-ink/5 px-4 py-2 font-display text-[14px] font-bold text-ink-soft">
   343→              {hits} correct · {run.picks.length - hits} missed
   344→            </span>
   345→          </div>
   346→
   347→          <Flashcard
   348→            key={card.id}
   349→            card={card}
   350→            onFlipRef={flipRef}
   351→            onJudge={judge}
   352→            onOpenTopic={onOpenTopic}
   353→          />
   354→
   355→          {/* combo quip */}
   356→          <AnimatePresence>
   357→            {quip && (
   358→              <motion.div
   359→                initial={{ opacity: 0, y: 16, scale: 0.9 }}
   360→                animate={{ opacity: 1, y: 0, scale: 1 }}
   361→                exit={{ opacity: 0, scale: 0.9 }}
   362→                className="mt-6 flex items-end justify-center gap-2"
   363→              >
   364→                <Mascot main="#FFD43B" mood="cheer" size={72} bubbles={false} />
   365→                <SpeechBubble className="mb-3" tailClass="left-[-9px] rotate-[135deg]">
   366→                  <span className="font-display text-[13.5px] font-semibold text-ink">{quip}</span>
   367→                </SpeechBubble>
   368→              </motion.div>
   369→            )}
   370→          </AnimatePresence>
   371→
   372→          <div className="relative flex justify-center">
   373→            <Confetti fire={confetti} />
   374→          </div>
   375→
   376→          <p className="mt-8 text-center text-[12.5px] font-bold text-ink-faint">
   377→            Space to flip · → got it · ← not yet
   378→          </p>
   379→        </section>
   380→      )}
   381→
   382→      {/* ————— DONE: results ————— */}
   383→      {phase === "done" && run && (
   384→        <section className="relative mt-10">
   385→          <Reveal>
   386→            <div
   387→              className="relative overflow-hidden rounded-[40px] p-7 text-center sm:p-10"
   388→              style={{ background: tint("#12B886", 9) }}
   389→            >
   390→              <Blob color={tint("#12B886", 18)} className="bottom-[-25%] left-[-10%] h-[65%] w-[50%]" />
   391→              <div className="relative z-10">
   392→                <Float>
   393→                  <Mascot
   394→                    main="#12B886"
   395→                    mood={hits / run.cards.length >= 0.7 ? "cheer" : "idle"}
   396→                    size={150}
   397→                  />
   398→                </Float>
   399→                <h2 className="mt-4 font-display text-[clamp(1.7rem,3.8vw,2.4rem)] font-semibold text-ink">
   400→                  {hits === run.cards.length
   401→                    ? "All correct."
   402→                    : hits / run.cards.length >= 0.7
   403→                      ? "Good work."
   404→                      : hits / run.cards.length >= 0.4
   405→                        ? "Keep practising."
   406→                        : "Worth another look."}
   407→                </h2>
   408→                <p className="mt-2 text-[15px] text-ink-soft">
   409→                  {hits} of {run.cards.length} correct · best streak {bestStreak}
   410→                  {bestStreak >= stats.best && bestStreak > 0 && (
   411→                    <span className="ml-1.5 inline-flex items-center gap-1 rounded-full bg-lemon/25 px-2.5 py-0.5 text-[11.5px] font-extrabold text-lemon">
   412→                      <Trophy className="h-3 w-3" /> new record
   413→                    </span>
   414→                  )}
   415→                </p>
   416→
   417→                {/* mini bars */}
   418→                <div className="mx-auto mt-6 flex max-w-sm items-center gap-3">
   419→                  <div className="h-5 flex-1 overflow-hidden rounded-full bg-paper/80">
   420→                    <motion.div
   421→                      className="h-full rounded-full bg-mint"
   422→                      initial={{ width: 0 }}
   423→                      animate={{ width: `${(hits / run.cards.length) * 100}%` }}
   424→                      transition={{ type: "spring", stiffness: 110, damping: 18, delay: 0.2 }}
   425→                    />
   426→                  </div>
   427→                  <span className="font-display text-[15px] font-bold text-ink">
   428→                    {Math.round((hits / run.cards.length) * 100)}%
   429→                  </span>
   430→                </div>
   431→
   432→                <div className="mt-8 flex flex-wrap items-center justify-center gap-3">
   433→                  <motion.button
   434→                    type="button"
   435→                    onClick={() => {
   436→                      setFiredMilestone(0);
   437→                      setPhase("idle");
   438→                      setRun(null);
   439→                    }}
   440→                    whileHover={{ y: -2 }}
   441→                    whileTap={{ scale: 0.95, y: 2 }}
   442→                    className="candy inline-flex min-h-[52px] cursor-pointer items-center gap-2 rounded-full px-7 py-3.5 font-display text-[16px] font-semibold text-white select-none"
   443→                    style={{ "--candy-bg": "#12B886", "--candy-side": shade("#12B886") } as React.CSSProperties}
   444→                  >
   445→                    <RotateCcw className="h-4.5 w-4.5" /> Shuffle again
   446→                  </motion.button>
   447→                  <SoftButton
   448→                    onClick={() => {
   449→                      setPhase("idle");
   450→                      setRun(null);
   451→                    }}
   452→                  >
   453→                    Change the deck
   454→                  </SoftButton>
   455→                </div>
   456→              </div>
   457→            </div>
   458→          </Reveal>
   459→
   460→          {/* the cards you missed */}
   461→          {run.picks.some((p) => p === "miss") && (
   462→            <Reveal delay={0.1} className="mt-8">
   463→              <div className="mb-3 font-display text-[13px] font-bold tracking-[0.16em] text-ink-faint uppercase">
   464→                cards to review
   465→              </div>
   466→              <div className="flex flex-col gap-3">
   467→                {run.cards.map((c, i) =>
   468→                  run.picks[i] === "miss" ? (
   469→                    <motion.button
   470→                      key={c.id}
   471→                      type="button"
   472→                      onClick={() => onOpenTopic(c.unit, c.topicId)}
   473→                      whileHover={{ y: -3, rotate: 0 }}
   474→                      whileTap={{ scale: 0.98 }}
   475→                      className="flex cursor-pointer items-start gap-3.5 rounded-[26px] bg-paper p-4.5 text-left puff sm:items-center sm:p-5"
   476→                      style={{ transform: `rotate(${i % 2 ? 0.5 : -0.5}deg)` }}
   477→                    >
   478→                      <span
   479→                        className="grid h-10 w-10 shrink-0 place-items-center rounded-2xl"
   480→                        style={{ background: tint(c.unitMain, 16) }}
   481→                      >
   482→                        <X className="h-5 w-5" style={{ color: shade(c.unitMain, 62) }} strokeWidth={3} />
   483→                      </span>
   484→                      <span className="min-w-0">
   485→                        <span className="flex flex-wrap items-center gap-2">
   486→                          <Chip main={c.unitMain}>{c.topicLabel}</Chip>
   487→                          <span className="text-[11.5px] font-extrabold text-ink-faint">
   488→                            {c.unit}
   489→                          </span>
   490→                        </span>
   491→                        <span className="mt-1 block text-[14.5px] leading-snug font-bold text-ink">
   492→                          <Fmt text={c.q} />
   493→                        </span>
   494→                      </span>
   495→                    </motion.button>
   496→                  ) : null
   497→                )}
   498→              </div>
   499→            </Reveal>
   500→          )}
   501→
   502→          {/* revised topics hint */}
   503→          <p className="mt-8 text-center text-[12.5px] font-bold text-ink-faint">
   504→            {revised.length === 0
   505→              ? "Tip: mark topics as revised from their topic pages."
   506→              : `${revised.length} topic${revised.length === 1 ? "" : "s"} marked as revised so far.`}
   507→          </p>
   508→        </section>
   509→      )}
   510→    </div>
   511→  );
   512→}

   514→/* ————— the flashcard itself: a fat 3D flip ————— */
   515→
   516→function Flashcard({
   517→  card,
   518→  onFlipRef,
   519→  onJudge,
   520→  onOpenTopic,
   521→}: {
   522→  card: Card;
   523→  onFlipRef: React.MutableRefObject<(() => void) | null>;
   524→  onJudge: (v: "hit" | "miss") => void;
   525→  onOpenTopic: (code: UnitCode, topicId: string) => void;
   526→}) {
   527→  const [flipped, setFlipped] = useState(false);
   528→  const [nudge, setNudge] = useState(0);
   529→
   530→  const flip = useCallback(() => {
   531→    setFlipped((f) => {
   532→      if (!f) setNudge((n) => n + 1);
   533→      return !f;
   534→    });
   535→  }, []);
   536→
   537→  useEffect(() => {
   538→    onFlipRef.current = flip;
   539→    return () => {
   540→      onFlipRef.current = null;
   541→    };
   542→  }, [flip, onFlipRef]);
   543→
   544→  return (
   545→    <div className="[perspective:1400px]">
   546→      {/* grid-stack: both faces share one cell, so the card grows to fit
   547→          whichever face is taller — long answers never spill out */}
   548→      <motion.div
   549→        className="relative mx-auto grid w-full max-w-xl [transform-style:preserve-3d]"
   550→        animate={{ rotateY: flipped ? 180 : 0, x: nudge ? [0, -6, 6, -3, 0] : 0 }}
   551→        transition={{
   552→          rotateY: { type: "spring", stiffness: 260, damping: 22 },
   553→          x: { duration: 0.4 },
   554→        }}
   555→        style={{ minHeight: 340 }}
   556→      >
   557→        {/* front — the question */}
   558→        <CardFace>
   559→          <div className="flex flex-wrap items-center gap-2">
   560→            <Chip main={card.unitMain}>{card.unit}</Chip>
   561→            <Chip>{card.topicLabel}</Chip>
   562→            <span className="ml-auto font-display text-[12px] font-bold tracking-[0.14em] text-ink-faint uppercase">
   563→              tap to flip
   564→            </span>
   565→          </div>
   566→          <p className="mt-7 font-display text-[clamp(1.3rem,3vw,1.7rem)] leading-[1.4] font-semibold text-ink">
   567→            <Fmt text={card.q} />
   568→          </p>
   569→          <div className="mt-auto pt-7">
   570→            <motion.button
   571→              type="button"
   572→              onClick={flip}
   573→              whileHover={{ y: -2 }}
   574→              whileTap={{ scale: 0.95, y: 2 }}
   575→              className="candy inline-flex min-h-[48px] cursor-pointer items-center gap-2 rounded-full px-6 py-3 font-display text-[15px] font-semibold text-white select-none"
   576→              style={
   577→                {
   578→                  "--candy-bg": card.unitMain,
   579→                  "--candy-side": shade(card.unitMain),
   580→                } as React.CSSProperties
   581→              }
   582→            >
   583→              <Sparkles className="h-4.5 w-4.5" /> Reveal the answer
   584→            </motion.button>
   585→          </div>
   586→        </CardFace>
   587→
   588→        {/* back — the answer */}
   589→        <CardFace back>
   590→          <div className="flex flex-wrap items-center gap-2">
   591→            <span className="inline-flex items-center gap-1.5 rounded-full bg-paper px-3.5 py-1.5 font-display text-[12px] font-bold text-ink">
   592→              <Check className="h-3.5 w-3.5 text-mint" strokeWidth={3} /> answer
   593→            </span>
   594→            <span className="text-[12px] font-extrabold text-ink-faint">
   595→              {card.topicTitle}
   596→            </span>
   597→          </div>
   598→          <p className="mt-6 text-[15.5px] leading-[1.8] text-ink">
   599→            <Fmt text={card.a} />
   600→          </p>
   601→          <div className="mt-auto flex flex-col gap-3 pt-7">
   602→            <div className="flex flex-wrap gap-3">
   603→              <motion.button
   604→                type="button"
   605→                onClick={() => onJudge("hit")}
   606→                whileHover={{ y: -2 }}
   607→                whileTap={{ scale: 0.95, y: 2 }}
   608→                className="candy inline-flex min-h-[48px] flex-1 cursor-pointer items-center justify-center gap-2 rounded-full px-5 py-3 font-display text-[15px] font-semibold text-white select-none"
   609→                style={{ "--candy-bg": "#12B886", "--candy-side": "#0ca678" } as React.CSSProperties}
   610→              >
   611→                <Check className="h-4.5 w-4.5" strokeWidth={3} /> Got it
   612→              </motion.button>
   613→              <motion.button
   614→                type="button"
   615→                onClick={() => onJudge("miss")}
   616→                whileHover={{ y: -2 }}
   617→                whileTap={{ scale: 0.95, y: 2 }}
   618→                className="candy inline-flex min-h-[48px] flex-1 cursor-pointer items-center justify-center gap-2 rounded-full px-5 py-3 font-display text-[15px] font-semibold text-white select-none"
   619→                style={{ "--candy-bg": "#FF6B6B", "--candy-side": "#e8590c" } as React.CSSProperties}
   620→              >
   621→                <X className="h-4.5 w-4.5" strokeWidth={3} /> Not yet
   622→              </motion.button>
   623→            </div>
   624→            <button
   625→              type="button"
   626→              onClick={() => onOpenTopic(card.unit, card.topicId)}
   627→              className="cursor-pointer text-[12.5px] font-extrabold text-ink-faint underline decoration-dotted underline-offset-4 hover:text-ink-soft"
   628→            >
   629→              reopen {card.topicLabel.toLowerCase()} · {card.topicTitle}
   630→            </button>
   631→          </div>
   632→        </CardFace>
   633→      </motion.div>
   634→
   635→      {/* the whole card is clickable on the front */}
   636→      {!flipped && (
   637→        <button
   638→          type="button"
   639→          aria-label="Flip card"
   640→          onClick={flip}
   641→          className="absolute inset-0 cursor-pointer rounded-[36px]"
   642→          style={{ zIndex: 5 }}
   643→        />
   644→      )}
   645→    </div>
   646→  );
   647→}

   652→      className="flex flex-col rounded-[36px] p-6 [grid-area:1/1] sm:p-8"

   662→    </div>
