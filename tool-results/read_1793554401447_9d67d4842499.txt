    52→/* ————— progress (localStorage) ————— */
    53→
    54→function useRevisedSpecs(docId: string) {
    55→  const [specs, setSpecs] = useState<string[]>(() => {
    56→    if (typeof window === "undefined") return [];
    57→    try {
    58→      return JSON.parse(window.localStorage.getItem(`rc-notes-${docId}`) ?? "[]");
    59→    } catch {
    60→      /* first visit */
    61→      return [];
    62→    }
    63→  });
    64→  const toggle = (spec: string) =>
    65→    setSpecs((prev) => {
    66→      const next = prev.includes(spec) ? prev.filter((s) => s !== spec) : [...prev, spec];
    67→      try {
    68→        window.localStorage.setItem(`rc-notes-${docId}`, JSON.stringify(next));
    69→      } catch {
    70→        /* private mode */
    71→      }
    72→      return next;
    73→    });
    74→  return { specs, toggle };
    75→}

    77→function scrollToSpec(spec: string) {
    78→  document.getElementById(`spec-${spec.replace(".", "-")}`)?.scrollIntoView({ behavior: "smooth", block: "start" });
    79→}

   315→            transition={{ type: "spring", stiffness: 320, damping: 15 }}
   316→            className="inline-flex min-h-[56px] cursor-pointer items-center gap-3 rounded-full px-7 py-4 font-display text-[16px] font-semibold"

   327→                </span>
   328→                Marked as revised
   329→              </>
   330→            ) : (
   331→              <>
   332→                <Sticker className="h-5.5 w-5.5" style={{ color: main }} />
   333→                Mark the whole note as revised
   334→              </>
   335→            )}
   336→          </motion.button>
   337→        </div>
   338→      </Reveal>
   339→      <p className="mt-3 text-center text-[12.5px] font-bold text-ink-faint">
   340→        {isRevised ? `All ${doc.sections.length} sections revised.` : `Revise all ${doc.sections.length} sections, then mark the note as done.`}
   341→      </p>
   342→
   343→      <p className="mt-10 text-center text-[12px] leading-relaxed font-bold text-ink-faint">
   344→        <Fmt text={`==${unit.code}== · ${unit.title} · text kept verbatim from the ${doc.source}`} />
   345→      </p>
   346→    </div>
   347→  );
   348→}

   486→          <div className="flex flex-wrap items-center gap-x-3 gap-y-2.5">
   487→            {b.eq.split(/\s{2,}/).map((part, i) => (
   488→              <span key={i} className="rounded-2xl bg-paper px-4 py-2.5 font-mono text-[14.5px] font-semibold text-ink shadow-[0_3px_0_rgb(61_43_31/0.06)]">
   489→                <Fmt text={part.trim()} />
   490→              </span>
   491→            ))}
   492→          </div>
   493→          {b.note && (
   494→            <p className="mt-3 text-[13px] leading-relaxed font-bold text-ink-soft">
   495→              <Fmt text={b.note} />
   496→            </p>
   497→          )}
   498→        </div>
   499→      );
   500→    case "table":
   501→      return <CandyTable head={b.head} rows={b.rows} caption={b.caption} main={main} />;
   502→    case "worked":
   503→      return <Worked title={b.title} question={b.question} steps={b.steps} answer={b.answer} main={main} />;
   504→    case "steps":
   505→      return <Journey caption={b.caption} items={b.items} main={main} />;
   506→    case "checkpoint":
   507→      return (
   508→        <div className="flex items-center gap-3 rounded-full bg-paper px-5 py-4 puff">
   509→          <span className="grid h-10 w-10 shrink-0 place-items-center rounded-full" style={{ background: tint("#12B886", 18) }}>
   510→            <BookOpen className="h-5 w-5 text-mint" />
   511→          </span>
   512→          <p className="text-[15px] font-bold text-ink">
   513→            Checkpoint — the check point questions for this section are on page {b.page} of the textbook.
   514→          </p>
   515→        </div>
   516→      );
   517→    case "figure":
   518→      return <Figure kind={b.kind} caption={b.caption} main={main} />;
   519→    case "wow":
   520→      return (
   521→        <div className="max-w-[72ch] rounded-[22px] p-5 sm:p-6" style={{ background: tint("#9775FA", 12) }}>
   522→          <div className="flex items-start gap-3">
   523→            <span className="grid h-9 w-9 shrink-0 place-items-center rounded-2xl bg-paper">
   524→              <Lightbulb className="h-5 w-5 text-grape" />
   525→            </span>
   526→            <div>
   527→              <div className="font-display text-[11px] font-extrabold tracking-[0.18em] text-grape uppercase">Key idea</div>
   528→              <p className="mt-1 text-[15px] leading-[1.75] text-ink">
   529→                <Fmt text={b.text} />
   530→              </p>
   531→            </div>
   532→          </div>
   533→        </div>
   534→      );
   535→    default:
   536→      return null;
   537→  }
   538→}

   648→const STEP_ICONS: Record<string, typeof Flame> = {
   649→  flame: Flame,

   651→  arrow: ArrowRight,

   656→
   657→function Journey({
   658→  caption,
   659→  items,
   660→  main,
   661→}: {
   662→  caption?: string;
   663→  items: { title: string; text: string; icon?: string }[];
   664→  main: string;
   665→}) {
   666→  const [open, setOpen] = useState(0);
   667→  return (
   668→    <div>
   669→      {caption && <p className="mb-3 text-[12px] font-extrabold tracking-[0.08em] text-ink-faint uppercase">{caption}</p>}
   670→      <div className="flex flex-col gap-2.5">
   671→        {items.map((it, i) => {
   672→          const Icon = STEP_ICONS[it.icon ?? "spark"] ?? Sparkles;
   673→          const isOpen = open === i;
   674→          return (
   675→            <motion.div key={i} layout className="rounded-[22px] bg-paper puff" animate={isOpen ? { rotate: 0 } : {}}>
   676→              <button
   677→                type="button"
   678→                onClick={() => setOpen(isOpen ? -1 : i)}
   679→                aria-expanded={isOpen}
   680→                className="flex w-full cursor-pointer items-center gap-3 px-4 py-3.5 text-left sm:px-5"
   681→              >
   682→                <span className="grid h-9 w-9 shrink-0 place-items-center rounded-full" style={{ background: isOpen ? main : tint(main, 14) }}>
   683→                  <Icon className="h-4.5 w-4.5" style={{ color: isOpen ? "#fff" : shade(main, 62) }} strokeWidth={2.5} />
   684→                </span>
   685→                <span className="min-w-0 flex-1 font-display text-[15.5px] font-semibold text-ink">{it.title}</span>
   686→                <motion.span animate={{ rotate: isOpen ? 180 : 0 }} transition={{ type: "spring", stiffness: 300, damping: 20 }}>
   687→                  <ChevronDown className="h-4.5 w-4.5 text-ink-faint" />
   688→                </motion.span>
   689→              </button>
   690→              <AnimatePresence initial={false}>
   691→                {isOpen && (
   692→                  <motion.div
   693→                    initial={{ opacity: 0, height: 0 }}
   694→                    animate={{ opacity: 1, height: "auto" }}
   695→                    exit={{ opacity: 0, height: 0 }}

   701→                    </p>
   702→                  </motion.div>
   703→                )}
   704→              </AnimatePresence>
   705→            </motion.div>
   706→          );
   707→        })}

   709→    </div>
   710→  );

   721→    case "spectrum-boron": node = <FigSpectrumBoron />; break;
   722→    case "spectrum-bromine": node = <FigSpectrumBromine />; break;
   723→    case "spectrum-cl2": node = <FigSpectrumCl2 />; break;
   724→    case "spectrum-ethanol": node = <FigSpectrumEthanol />; break;
   725→    case "orbitals": node = <FigOrbitals main={main} />; break;
   726→    case "fill-demo": node = <FigFillDemo main={main} />; break;
   727→    case "fill-order": node = <FigFillOrder main={main} />; break;
   728→    case "boxes-c": node = <FigBoxesC main={main} />; break;
   729→    case "boxes-s": node = <FigBoxesS main={main} />; break;
   730→    case "ie-graph": node = <FigIeGraph />; break;

   743→        </p>

   745→    </div>
   746→  );
