    48→    48→   Each section is a full-width tonal band: most borrow a candy

    75→    75→export function NotesReader({ doc, unit, onBack }: { doc: NotesDoc; unit: Unit; onBack: () => void }) {
    76→    76→  const main = unit.candy.main;
    77→    77→  const [active, setActive] = useState<string>(doc.sections[0]?.spec ?? "");
    78→    78→
    79→    79→  const { scrollYProgress } = useScroll();
    80→    80→  const scaleX = useSpring(scrollYProgress, { stiffness: 140, damping: 26, mass: 0.4 });
    81→    81→
    82→    82→  /* scroll spy */
    83→    83→  const spy = useRef<IntersectionObserver | null>(null);
    84→    84→  useEffect(() => {
    85→    85→    spy.current = new IntersectionObserver(
    86→    86→      (entries) => {
    87→    87→        for (const e of entries) {
    88→    88→          if (e.isIntersecting) setActive(e.target.id.replace("spec-", "").replace("-", "."));
    89→    89→        }
    90→    90→      },
    91→    91→      { rootMargin: "-30% 0px -60% 0px" }
    92→    92→    );
    93→    93→    for (const s of doc.sections) {
    94→    94→      const el = document.getElementById(`spec-${s.spec.replace(".", "-")}`);
    95→    95→      if (el) spy.current?.observe(el);
    96→    96→    }
    97→    97→    return () => spy.current?.disconnect();
    98→    98→  }, [doc]);
    99→    99→
   100→   100→  const count = (k: NotesBlock["k"]) => {
   101→   101→    let n = 0;
   102→   102→    const walk = (blocks: NotesBlock[]) => blocks.forEach((b) => b.k === k && n++);
   103→   103→    walk(doc.intro);
   104→   104→    doc.sections.forEach((s) => walk(s.blocks));

   110→   110→    `${count("worked")} worked examples`,

   115→   115→  const accentOf = (i: number) => BANDS[i % BANDS.length];
   116→   116→  const bandOf = (i: number) => tint(accentOf(i), 10);
   117→   117→  /* every third room sits on the deep roast — a dark interlude

   120→   120→  const bgOf = (i: number) => (isDarkBand(i) ? "var(--color-cocoa)" : bandOf(i));

   132→   132→
   133→   133→      {/* ————— header ————— */}

   135→   135→        <PopIn>
   136→   136→          <SoftButton onClick={onBack} className="min-h-[42px] px-4 py-2 text-[13.5px]" ariaLabel="Back to the topic page">

   139→   139→        </PopIn>
   140→   140→
   141→   141→        <PopIn delay={0.05} className="mt-7 flex flex-wrap items-center gap-2">
   142→   142→          <span
   143→   143→            className="inline-flex -rotate-1 items-center gap-1.5 rounded-full px-3.5 py-1.5 font-display text-[11.5px] font-bold tracking-[0.14em] text-white uppercase"

   145→   145→          >
   146→   146→            <BookOpen className="h-3.5 w-3.5" /> Revision note
   147→   147→          </span>
   148→   148→          <span className="inline-flex items-center rounded-full bg-ink/5 px-3.5 py-1.5 text-[12.5px] font-bold text-ink-soft">
   149→   149→            {doc.source}
   150→   150→          </span>
   151→   151→        </PopIn>
   152→   152→        <PopIn delay={0.1}>
   153→   153→          <p className="mt-6 font-display text-[15px] font-semibold tracking-[0.06em] uppercase" style={{ color: shade(main, 62) }}>
   154→   154→            {doc.kicker}

   167→   167→              <p className="text-[14.5px] leading-[1.75] text-ink">
   168→   168→                {doc.intro.map((b, i) =>

   183→   183→              <Sparkles className="h-3.5 w-3.5" style={{ color: main }} /> {s}
   184→   184→            </span>

   186→   186→        </PopIn>
   187→   187→
   188→   188→        <Squiggle color={main} className="mt-7 h-3.5 w-full opacity-60" />
   189→   189→      </header>
   190→   190→
   191→   191→      {/* ————— contents ————— */}
   192→   192→      <Reveal className="mx-auto max-w-3xl px-4 pt-4 sm:px-8">

   194→   194→          <div className="flex items-center gap-3">
   195→   195→            <span className="grid h-10 w-10 shrink-0 place-items-center rounded-2xl" style={{ background: tint(main, 16) }}>
   196→   196→              <Target className="h-5 w-5" style={{ color: shade(main, 62) }} />
   197→   197→            </span>
   198→   198→            <div>
   199→   199→              <h2 className="font-display text-[18px] font-semibold text-ink">Contents</h2>

   201→   201→            </div>
   202→   202→          </div>
   203→   203→          <div className="mt-4 grid grid-cols-2 gap-2 sm:grid-cols-3">
   204→   204→            {doc.sections.map((s, i) => {
   205→   205→              const accent = accentOf(i);
   206→   206→              return (

   212→   212→                    isDarkBand(i) ? " band-dark" : ""

   214→   214→                  style={isDarkBand(i) ? undefined : { background: bandOf(i) }}
   215→   215→                >
   216→   216→                  <span
   217→   217→                    className="grid h-6 w-6 shrink-0 place-items-center rounded-full font-display text-[10.5px] font-bold"
   218→   218→                    style={{ background: accent, color: readableInk(accent) }}
   219→   219→                  >
   220→   220→                    {s.spec}
   221→   221→                  </span>
   222→   222→                  <span className="truncate text-[12px] font-bold text-ink">{s.title}</span>
   223→   223→                </button>
   224→   224→              );
   225→   225→            })}
   226→   226→          </div>
   227→   227→        </div>
   228→   228→      </Reveal>
   229→   229→
   230→   230→      {/* ————— the string of rooms ————— */}
   231→   231→      <div className="mt-10">
   232→   232→        {/* sticky rail rides above every band */}
   233→   233→        <div className="sticky top-[76px] z-40 mx-auto max-w-3xl px-4 sm:px-8">

   235→   235→            className="rounded-[24px] bg-paper/95 px-3.5 py-2.5 backdrop-blur-md"

   237→   237→          >
   238→   238→            <div className="flex items-center gap-3">
   239→   239→              {activeSection ? (
   240→   240→                <span className="min-w-0 flex-1 truncate font-display text-[13.5px] font-semibold text-ink">
   241→   241→                  <span style={{ color: shade(accentOf(activeIndex), 45) }}>{activeSection.spec}</span> · {activeSection.title}
   242→   242→                </span>
   243→   243→              ) : (
   244→   244→                <span className="min-w-0 flex-1 truncate font-display text-[13.5px] font-semibold text-ink">{doc.title}</span>
   245→   245→              )}
   246→   246→            </div>
   247→   247→            <div className="no-scrollbar mt-2 flex items-center gap-1.5 overflow-x-auto">
   248→   248→              {doc.sections.map((s, i) => {
   249→   249→                const isActive = active === s.spec;
   250→   250→                const accent = accentOf(i);

   252→   252→                  <motion.button
   253→   253→                    key={s.spec}
   254→   254→                    type="button"
   255→   255→                    whileTap={{ scale: 0.88 }}
   256→   256→                    onClick={() => scrollToSpec(s.spec)}
   257→   257→                    className="grid h-8 min-w-8 shrink-0 cursor-pointer place-items-center rounded-full px-2 font-display text-[11px] font-bold"
   258→   258→                    style={
   259→   259→                      isActive
   260→   260→                        ? { background: accent, color: readableInk(accent), boxShadow: `0 3px 0 ${shade(accent)}` }
   261→   261→                        : { background: tint(accent, 22), color: shade(accent, 62) }
   262→   262→                    }
   263→   263→                    title={`${s.spec} — ${s.title}`}

   275→   275→        <div className="mt-8">

   277→   277→          {doc.sections.map((s, i) => {

   279→   279→            const dark = isDarkBand(i);
   280→   280→            const last = i === doc.sections.length - 1;
   281→   281→            return (
   282→   282→              <Fragment key={s.spec}>
   283→   283→                <section
   284→   284→                  id={`spec-${s.spec.replace(".", "-")}`}
   285→   285→                  className={`relative scroll-mt-[190px]${dark ? " band-dark" : ""}`}
   286→   286→                  style={dark ? undefined : { background: bandOf(i) }}
   287→   287→                >
   288→   288→                  <div className="mx-auto max-w-3xl px-4 py-14 sm:px-8 sm:py-16">
   289→   289→                    <SpecSectionView section={s} accent={accent} />
   290→   290→                  </div>
   291→   291→                </section>
   292→   292→                <BandWave from={bgOf(i)} to={last ? "var(--color-cream)" : bgOf(i + 1)} />
   293→   293→              </Fragment>
   294→   294→            );
   295→   295→          })}
   296→   296→        </div>
   297→   297→      </div>
   298→   298→
   299→   299→      {/* ————— end of the note ————— */}
   300→   300→      <div className="mx-auto max-w-3xl px-4 pt-10 pb-24 text-center sm:px-8">
   301→   301→        <Reveal className="flex flex-col items-center gap-5">
   302→   302→          <Squiggle color={main} className="h-3 w-40 opacity-70" />
   303→   303→          <p className="font-display text-[19px] font-semibold text-ink">
   304→   304→            That&apos;s the whole note — every spec point, word for word.
   305→   305→          </p>
   306→   306→          <SoftButton onClick={onBack} ariaLabel="Back to the topic page">
   307→   307→            <ArrowLeft className="h-4.5 w-4.5" /> Back to the topic page
   308→   308→          </SoftButton>
   309→   309→          <p className="mt-2 text-[12px] leading-relaxed font-bold text-ink-faint">
   310→   310→            <Fmt text={`==${unit.code}== · ${unit.title} · text kept verbatim from the ${doc.source}`} />
   311→   311→          </p>
   312→   312→        </Reveal>
   313→   313→      </div>
   314→   314→    </div>
   315→   315→  );
   316→   316→}
