    47→   Reaction Club · notes reader — one enclosed room per spec point.
    48→   Each section is a full-width tonal band in its own colour, and
    49→   the footer's wave rides between the bands, so the note reads as

    72→/* ————— main reader ————— */
    73→
    74→export function NotesReader({ doc, unit, onBack }: { doc: NotesDoc; unit: Unit; onBack: () => void }) {
    75→  const main = unit.candy.main;
    76→  const [active, setActive] = useState<string>(doc.sections[0]?.spec ?? "");
    77→
    78→  const { scrollYProgress } = useScroll();
    79→  const scaleX = useSpring(scrollYProgress, { stiffness: 140, damping: 26, mass: 0.4 });
    80→
    81→  /* scroll spy */
    82→  const spy = useRef<IntersectionObserver | null>(null);
    83→  useEffect(() => {
    84→    spy.current = new IntersectionObserver(
    85→      (entries) => {
    86→        for (const e of entries) {
    87→          if (e.isIntersecting) setActive(e.target.id.replace("spec-", "").replace("-", "."));
    88→        }
    89→      },
    90→      { rootMargin: "-30% 0px -60% 0px" }
    91→    );
    92→    for (const s of doc.sections) {
    93→      const el = document.getElementById(`spec-${s.spec.replace(".", "-")}`);
    94→      if (el) spy.current?.observe(el);
    95→    }
    96→    return () => spy.current?.disconnect();
    97→  }, [doc]);
    98→
    99→  const count = (k: NotesBlock["k"]) => {
   100→    let n = 0;
   101→    const walk = (blocks: NotesBlock[]) => blocks.forEach((b) => b.k === k && n++);
   102→    walk(doc.intro);
   103→    doc.sections.forEach((s) => walk(s.blocks));
   104→    if (doc.outro) walk(doc.outro);
   105→    return n;
   106→  };
   107→  const stats = [
   108→    `${doc.sections.length} spec sections`,
   109→    `${count("worked")} worked examples`,
   110→    `${count("figure")} diagrams`,
   111→    `${count("checkpoint")} textbook checkpoints`,
   112→  ];
   113→
   114→  const accentOf = (i: number) => BANDS[i % BANDS.length];
   115→  const bandOf = (i: number) => tint(accentOf(i), 10);
   116→  const activeIndex = Math.max(0, doc.sections.findIndex((s) => s.spec === active));
   117→  const activeSection = doc.sections[activeIndex];
   118→
   119→  return (
   120→    <div className="relative">
   121→      {/* reading progress */}
   122→      <motion.div
   123→        aria-hidden
   124→        className="fixed top-0 left-0 z-[70] h-[7px] w-full origin-left rounded-r-full"
   125→        style={{ scaleX, background: `linear-gradient(90deg, ${main}, #FFD43B)` }}
   126→      />
   127→
   128→      {/* ————— header ————— */}
   129→      <header className="mx-auto max-w-3xl px-4 pt-24 sm:px-8">
   130→        <PopIn>
   131→          <SoftButton onClick={onBack} className="min-h-[42px] px-4 py-2 text-[13.5px]" ariaLabel="Back to the topic page">
   132→            <ArrowLeft className="h-4 w-4" /> Back to the topic page
   133→          </SoftButton>
   134→        </PopIn>
   135→
   136→        <PopIn delay={0.05} className="mt-7 flex flex-wrap items-center gap-2">
   137→          <span
   138→            className="inline-flex -rotate-1 items-center gap-1.5 rounded-full px-3.5 py-1.5 font-display text-[11.5px] font-bold tracking-[0.14em] text-white uppercase"
   139→            style={{ background: main }}
   140→          >
   141→            <BookOpen className="h-3.5 w-3.5" /> Revision note
   142→          </span>
   143→          <span className="inline-flex items-center rounded-full bg-ink/5 px-3.5 py-1.5 text-[12.5px] font-bold text-ink-soft">
   144→            {doc.source}
   145→          </span>
   146→        </PopIn>
   147→        <PopIn delay={0.1}>
   148→          <p className="mt-6 font-display text-[15px] font-semibold tracking-[0.06em] uppercase" style={{ color: shade(main, 62) }}>
   149→            {doc.kicker}
   150→          </p>
   151→          <h1 className="mt-2 font-display text-[clamp(2.1rem,5vw,3.2rem)] leading-[1.05] font-semibold tracking-tight text-ink">
   152→            {doc.title}
   153→          </h1>
   154→        </PopIn>
   155→
   156→        <div className="mt-6 flex flex-col items-start gap-5 sm:flex-row sm:items-center">
   157→          <div className="shrink-0">
   158→            <Mascot main={main} size={104} bubbles={false} />
   159→          </div>
   160→          <Reveal delay={0.15} className="min-w-0">
   161→            <SpeechBubble tailClass="left-8 -top-[7px] rotate-[225deg]">
   162→              <p className="text-[14.5px] leading-[1.75] text-ink">
   163→                {doc.intro.map((b, i) =>
   164→                  b.k === "p" ? (
   165→                    <span key={i} className="block">
   166→                      <Fmt text={b.text} />
   167→                    </span>
   168→                  ) : null
   169→                )}
   170→              </p>
   171→            </SpeechBubble>
   172→          </Reveal>
   173→        </div>
   174→
   175→        <PopIn delay={0.18} className="mt-6 flex flex-wrap gap-2">
   176→          {stats.map((s) => (
   177→            <span key={s} className="inline-flex items-center gap-1.5 rounded-full px-3 py-1.5 text-[12px] font-bold text-ink" style={{ background: tint(main, 12) }}>
   178→              <Sparkles className="h-3.5 w-3.5" style={{ color: main }} /> {s}
   179→            </span>
   180→          ))}
   181→        </PopIn>
   182→
   183→        <Squiggle color={main} className="mt-7 h-3.5 w-full opacity-60" />
   184→      </header>
   185→
   186→      {/* ————— contents ————— */}
   187→      <Reveal className="mx-auto max-w-3xl px-4 pt-4 sm:px-8">
   188→        <div className="rounded-[30px] bg-paper p-5 puff sm:p-6">
   189→          <div className="flex items-center gap-3">
   190→            <span className="grid h-10 w-10 shrink-0 place-items-center rounded-2xl" style={{ background: tint(main, 16) }}>
   191→              <Target className="h-5 w-5" style={{ color: shade(main, 62) }} />
   192→            </span>
   193→            <div>
   194→              <h2 className="font-display text-[18px] font-semibold text-ink">Contents</h2>
   195→              <p className="text-[12.5px] font-bold text-ink-faint">{doc.sections.length} sections · tap one to jump</p>
   196→            </div>
   197→          </div>
   198→          <div className="mt-4 grid grid-cols-2 gap-2 sm:grid-cols-3">
   199→            {doc.sections.map((s, i) => {
   200→              const accent = accentOf(i);
   201→              return (
   202→                <button
   203→                  key={s.spec}
   204→                  type="button"
   205→                  onClick={() => scrollToSpec(s.spec)}
   206→                  className="group flex cursor-pointer items-center gap-2 rounded-[18px] px-3 py-2.5 text-left transition-transform hover:-translate-y-0.5"
   207→                  style={{ background: bandOf(i) }}
   208→                >
   209→                  <span
   210→                    className="grid h-6 w-6 shrink-0 place-items-center rounded-full font-display text-[10.5px] font-bold"
   211→                    style={{ background: accent, color: readableInk(accent) }}
   212→                  >
   213→                    {s.spec}
   214→                  </span>
   215→                  <span className="truncate text-[12px] font-bold text-ink">{s.title}</span>
   216→                </button>
   217→              );
   218→            })}
   219→          </div>
   220→        </div>
   221→      </Reveal>
   222→
   223→      {/* ————— the string of rooms ————— */}
   224→      <div className="mt-10">
   225→        {/* sticky rail rides above every band */}
   226→        <div className="sticky top-[76px] z-40 mx-auto max-w-3xl px-4 sm:px-8">
   227→          <div
   228→            className="rounded-[24px] bg-paper/95 px-3.5 py-2.5 backdrop-blur-md"
   229→            style={{ boxShadow: "0 3px 0 var(--edge), 0 16px 34px -20px rgb(0 0 0 / 0.4)" }}
   230→          >
   231→            <div className="flex items-center gap-3">
   232→              {activeSection ? (
   233→                <span className="min-w-0 flex-1 truncate font-display text-[13.5px] font-semibold text-ink">
   234→                  <span style={{ color: shade(accentOf(activeIndex), 45) }}>{activeSection.spec}</span> · {activeSection.title}
   235→                </span>
   236→              ) : (
   237→                <span className="min-w-0 flex-1 truncate font-display text-[13.5px] font-semibold text-ink">{doc.title}</span>
   238→              )}
   239→            </div>
   240→            <div className="no-scrollbar mt-2 flex items-center gap-1.5 overflow-x-auto">
   241→              {doc.sections.map((s, i) => {
   242→                const isActive = active === s.spec;
   243→                const accent = accentOf(i);
   244→                return (
   245→                  <motion.button
   246→                    key={s.spec}
   247→                    type="button"
   248→                    whileTap={{ scale: 0.88 }}
   249→                    onClick={() => scrollToSpec(s.spec)}
   250→                    className="grid h-8 min-w-8 shrink-0 cursor-pointer place-items-center rounded-full px-2 font-display text-[11px] font-bold"
   251→                    style={
   252→                      isActive
   253→                        ? { background: accent, color: readableInk(accent), boxShadow: `0 3px 0 ${shade(accent)}` }
   254→                        : { background: tint(accent, 22), color: shade(accent, 62) }
   255→                    }
   256→                    title={`${s.spec} — ${s.title}`}
   257→                    aria-label={`Jump to spec ${s.spec}`}
   258→                    aria-current={isActive ? "true" : undefined}
   259→                  >
   260→                    {s.spec}
   261→                  </motion.button>
   262→                );
   263→              })}
   264→            </div>
   265→          </div>
   266→        </div>
   267→
   268→        <div className="mt-8">
   269→          <BandWave from="var(--color-cream)" to={bandOf(0)} />
   270→          {doc.sections.map((s, i) => {
   271→            const accent = accentOf(i);
   272→            const last = i === doc.sections.length - 1;
   273→            return (
   274→              <Fragment key={s.spec}>
   275→                <section id={`spec-${s.spec.replace(".", "-")}`} className="relative scroll-mt-[190px]" style={{ background: bandOf(i) }}>
   276→                  <div className="mx-auto max-w-3xl px-4 py-14 sm:px-8 sm:py-16">
   277→                    <SpecSectionView section={s} accent={accent} />
   278→                  </div>
   279→                </section>
   280→                <BandWave from={bandOf(i)} to={last ? "var(--color-cream)" : bandOf(i + 1)} />
   281→              </Fragment>
   282→            );
   283→          })}
   284→        </div>
   285→      </div>
   286→
   287→      {/* ————— end of the note ————— */}
   288→      <div className="mx-auto max-w-3xl px-4 pt-10 pb-24 text-center sm:px-8">
   289→        <Reveal className="flex flex-col items-center gap-5">
   290→          <Squiggle color={main} className="h-3 w-40 opacity-70" />
   291→          <p className="font-display text-[19px] font-semibold text-ink">
   292→            That&apos;s the whole note — every spec point, word for word.
   293→          </p>
   294→          <SoftButton onClick={onBack} ariaLabel="Back to the topic page">
   295→            <ArrowLeft className="h-4.5 w-4.5" /> Back to the topic page
   296→          </SoftButton>
   297→          <p className="mt-2 text-[12px] leading-relaxed font-bold text-ink-faint">
   298→            <Fmt text={`==${unit.code}== · ${unit.title} · text kept verbatim from the ${doc.source}`} />
   299→          </p>
   300→        </Reveal>
   301→      </div>
   302→    </div>
   303→  );
   304→}

   346→/* ————— block dispatcher ————— */
   347→
   348→function BlockView({ block: b, accent }: { block: NotesBlock; accent: string }) {
   349→  switch (b.k) {
   350→    case "p":
   351→      return (
   352→        <p className="max-w-[68ch] text-[16.5px] leading-[1.8] text-ink/90">
   353→          <Fmt text={b.text} />
   354→        </p>
   355→      );
   356→    case "h":
   357→      return (
   358→        <div className="flex items-center gap-2.5 pt-2">
   359→          <span className="h-2.5 w-2.5 shrink-0 rounded-full" style={{ background: accent }} />
   360→          <h3 className="font-display text-[21px] font-semibold tracking-tight text-ink">
   361→            <Fmt text={b.text} />
   362→          </h3>
   363→        </div>
   364→      );
   365→    case "bullets":
   366→      return (
   367→        <ul className="flex max-w-[70ch] flex-col gap-2.5">
   368→          {b.items.map((t, i) => (
   369→            <li key={i} className="flex items-start gap-3 text-[16px] leading-[1.8] text-ink/90">
   370→              <span className="mt-[11px] h-2 w-2 shrink-0 rounded-full" style={{ background: tint(accent, 55) }} />
   371→              <span>
   372→                <Fmt text={t} />
   373→              </span>
   374→            </li>
   375→          ))}
   376→        </ul>
   377→      );
   378→    case "note":
   379→      return (
   380→        <div className="max-w-[72ch] rounded-[20px] px-4 py-3.5" style={{ background: tint("#FFD43B", 18) }}>
   381→          <div className="flex items-start gap-3">
   382→            <StickyNote className="mt-0.5 h-4.5 w-4.5 shrink-0 text-[#B08D00]" />
   383→            <p className="text-[14.5px] leading-[1.7] text-ink">
   384→              <span className="mr-1.5 font-display text-[11.5px] font-extrabold tracking-[0.12em] text-[#B08D00] uppercase">{b.label}:</span>
   385→              <Fmt text={b.text} />
   386→            </p>
   387→          </div>
   388→        </div>
   389→      );
   390→    case "def":
   391→      return (
   392→        <div className="max-w-[72ch] rounded-[20px] bg-paper px-5 py-4 puff" style={{ borderLeft: `5px solid ${tint(accent, 40)}` }}>
   393→          <h4 className="font-display text-[16.5px] font-semibold text-ink">
   394→            <Fmt text={b.term} />
   395→          </h4>
   396→          <p className="mt-1.5 text-[15.5px] leading-[1.75] text-ink-soft">
   397→            <Fmt text={b.text} />
   398→          </p>
   399→        </div>
   400→      );
   401→    case "eq":
   402→      return (
   403→        <div className="max-w-[72ch] rounded-[22px] p-5" style={{ background: tint(accent, 9) }}>
   404→          <div className="flex flex-wrap items-center gap-x-3 gap-y-2.5">
   405→            {b.eq.split(/\s{2,}/).map((part, i) => (
   406→              <span key={i} className="rounded-2xl bg-paper px-4 py-2.5 font-mono text-[14.5px] font-semibold text-ink shadow-[0_3px_0_color-mix(in_oklab,var(--color-ink)_7%,transparent)]">
   407→                <Fmt text={part.trim()} />
   408→              </span>
   409→            ))}
   410→          </div>
   411→          {b.note && (
   412→            <p className="mt-3 text-[13px] leading-relaxed font-bold text-ink-soft">
   413→              <Fmt text={b.note} />
   414→            </p>
   415→          )}
   416→        </div>
   417→      );
   418→    case "table":
   419→      return <CandyTable head={b.head} rows={b.rows} caption={b.caption} accent={accent} />;
   420→    case "worked":
   421→      return <Worked title={b.title} question={b.question} steps={b.steps} answer={b.answer} accent={accent} />;
   422→    case "steps":
   423→      return <Journey caption={b.caption} items={b.items} accent={accent} />;
   424→    case "checkpoint":
   425→      return (
   426→        <div className="flex items-center gap-3 rounded-full bg-paper px-5 py-4 puff">
   427→          <span className="grid h-10 w-10 shrink-0 place-items-center rounded-full" style={{ background: tint("#12B886", 18) }}>
   428→            <BookOpen className="h-5 w-5 text-mint" />
   429→          </span>
   430→          <p className="text-[15px] font-bold text-ink">
   431→            Checkpoint — the check point questions for this section are on page {b.page} of the textbook.
   432→          </p>
   433→        </div>
   434→      );
   435→    case "figure":
   436→      return <Figure kind={b.kind} caption={b.caption} accent={accent} />;
   437→    case "wow":
   438→      return (
   439→        <div className="max-w-[72ch] rounded-[22px] p-5 sm:p-6" style={{ background: tint("#9775FA", 12) }}>
   440→          <div className="flex items-start gap-3">
   441→            <span className="grid h-9 w-9 shrink-0 place-items-center rounded-2xl bg-paper">
   442→              <Lightbulb className="h-5 w-5 text-grape" />
   443→            </span>
   444→            <div>
   445→              <div className="font-display text-[11px] font-extrabold tracking-[0.18em] text-grape uppercase">Key idea</div>
   446→              <p className="mt-1 text-[15px] leading-[1.75] text-ink">
   447→                <Fmt text={b.text} />
   448→              </p>
   449→            </div>
   450→          </div>
   451→        </div>
   452→      );
   453→    default:
   454→      return null;
   455→  }
   456→}

   458→/* ————— candy table ————— */
   459→
   460→function CandyTable({
   461→  head,
   462→  rows,
   463→  caption,
   464→  accent,
   465→}: {
   466→  head: string[];
   467→  rows: string[][];
   468→  caption?: string;

   471→  return (
