    55→/* ═══════════════════════════════════════════════════════════════
    56→   Reaction Club · notes reader — one enclosed room per spec point.
    57→   Each section is a full-width tonal band borrowing one candy
    58→   colour, walking the hue wheel so neighbours always contrast.
    59→   In the hybrid lab every third band is a deep cocoa room with
    60→   milk-toned fonts (see html.hybrid .room in globals.css); in the
    61→   other two labs the same bands stay candy — the class is inert.
    62→   The footer's wave rides between the bands, so the note reads as
    63→   a string of connected rooms rather than one long scroll.
    64→   ═══════════════════════════════════════════════════════════════ */
    65→
    66→/* the rotating palette — hue-wheel order: red → orange → sun →
    67→   green → purple → amber, so adjacent bands always differ (pink
    68→   stays reserved for WCH15's own identity) */
    69→const BANDS = ["#FF6B6B", "#FF922B", "#FFD43B", "#12B886", "#9775FA", "#F59F00"];
    70→
    71→const WAVE_D = "M0,52 C160,92 340,8 540,30 C740,52 900,96 1100,68 C1260,46 1370,22 1440,40 L1440,90 L0,90 Z";

    73→function BandWave({ from, to }: { from: string; to: string }) {

    77→        {/* fill rides on style so var() strings with fallbacks resolve */}
    78→        <path d={WAVE_D} style={{ fill: to }} />
    79→      </svg>
    80→    </div>

    94→/* ————— main reader ————— */
    95→
    96→export function NotesReader({ doc, unit, onBack }: { doc: NotesDoc; unit: Unit; onBack: () => void }) {
    97→  const main = unit.candy.main;
    98→  const [active, setActive] = useState<string>(doc.sections[0]?.spec ?? "");
    99→
   100→  const { scrollYProgress } = useScroll();
   101→  const scaleX = useSpring(scrollYProgress, { stiffness: 140, damping: 26, mass: 0.4 });
   102→
   103→  /* scroll spy */
   104→  const spy = useRef<IntersectionObserver | null>(null);
   105→  useEffect(() => {
   106→    spy.current = new IntersectionObserver(
   107→      (entries) => {
   108→        for (const e of entries) {
   109→          if (e.isIntersecting) setActive(e.target.id.replace("spec-", "").replace("-", "."));
   110→        }
   111→      },
   112→      { rootMargin: "-30% 0px -60% 0px" }
   113→    );
   114→    for (const s of doc.sections) {
   115→      const el = document.getElementById(`spec-${s.spec.replace(".", "-")}`);
   116→      if (el) spy.current?.observe(el);
   117→    }
   118→    return () => spy.current?.disconnect();
   119→  }, [doc]);
   120→
   121→  const count = (k: NotesBlock["k"]) => {
   122→    let n = 0;
   123→    const walk = (blocks: NotesBlock[]) => blocks.forEach((b) => b.k === k && n++);
   124→    walk(doc.intro);
   125→    doc.sections.forEach((s) => walk(s.blocks));
   126→    if (doc.outro) walk(doc.outro);
   127→    return n;
   128→  };
   129→  const stats = [
   130→    `${doc.sections.length} spec sections`,
   131→    `${count("worked")} worked examples`,
   132→    `${count("figure")} diagrams`,
   133→    `${count("checkpoint")} textbook checkpoints`,
   134→  ];
   135→
   136→  /* sticky spec rail — the chip strip scrolls sideways; gradient fades
   137→     hint at the cut-off chips on whichever side is hiding more */
   138→  const railRef = useRef<HTMLDivElement | null>(null);
   139→  const [railCut, setRailCut] = useState({ l: false, r: false });
   140→  const measureRail = () => {
   141→    const el = railRef.current;
   142→    if (!el) return;
   143→    setRailCut({
   144→      l: el.scrollLeft > 4,
   145→      r: el.scrollLeft + el.clientWidth < el.scrollWidth - 4,
   146→    });
   147→  };
   148→  useEffect(() => {
   149→    measureRail();
   150→    window.addEventListener("resize", measureRail);
   151→    return () => window.removeEventListener("resize", measureRail);
   152→    // eslint-disable-next-line react-hooks/exhaustive-deps
   153→  }, [doc]);
   154→
   155→  const accentOf = (i: number) => BANDS[i % BANDS.length];
   156→  /* bands mix their candy over --tint-base at --band-mix — 10% by
   157→     day, 14% after dark, so roast-tone neighbours stay distinct */
   158→  const bandOf = (i: number) =>
   159→    `color-mix(in oklab, ${accentOf(i)} var(--band-mix, 10%), var(--tint-base))`;
   160→  /* every third room goes cocoa in the hybrid lab (2.2, 2.5, 2.8 …).
   161→     Elsewhere --room-surface is undefined and the fallback keeps the
   162→     plain candy band — one expression, correct in all three labs. */
   163→  const isRoom = (i: number) => i % 3 === 1;
   164→  const bgOf = (i: number) => (isRoom(i) ? `var(--room-surface, ${bandOf(i)})` : bandOf(i));
   165→  const activeIndex = Math.max(0, doc.sections.findIndex((s) => s.spec === active));
   166→  const activeSection = doc.sections[activeIndex];
   167→
   168→  return (
   169→    <div className="relative">
   170→      {/* reading progress */}
   171→      <motion.div
   172→        aria-hidden
   173→        className="fixed top-0 left-0 z-[70] h-[7px] w-full origin-left rounded-r-full"
   174→        style={{ scaleX, background: `linear-gradient(90deg, ${main}, #FFD43B)` }}
   175→      />
   176→
   177→      {/* ————— header ————— */}
   178→      <header className="mx-auto max-w-3xl px-4 pt-24 sm:px-8">
   179→        <PopIn>
   180→          <SoftButton onClick={onBack} className="min-h-[42px] px-4 py-2 text-[13.5px]" ariaLabel="Back to the topic page">
   181→            <ArrowLeft className="h-4 w-4" /> Back to the topic page
   182→          </SoftButton>
   183→        </PopIn>
   184→
   185→        <PopIn delay={0.05} className="mt-7 flex flex-wrap items-center gap-2">
   186→          <span
   187→            className="inline-flex -rotate-1 items-center gap-1.5 rounded-full px-3.5 py-1.5 font-display text-[11.5px] font-bold tracking-[0.14em] text-white uppercase"
   188→            style={{ background: main }}
   189→          >
   190→            <BookOpen className="h-3.5 w-3.5" /> Revision note
   191→          </span>
   192→          <span className="inline-flex items-center rounded-full bg-ink/5 px-3.5 py-1.5 text-[12.5px] font-bold capitalize text-ink-soft">
   193→            {doc.source}
   194→          </span>
   195→        </PopIn>
   196→        <PopIn delay={0.1}>
   197→          <p className="mt-6 font-display text-[15px] font-semibold tracking-[0.06em] uppercase" style={{ color: inkMix(main, 52) }}>
   198→            {doc.kicker}
   199→          </p>
   200→          <h1 className="mt-2 font-display text-[clamp(2.1rem,5vw,3.2rem)] leading-[1.05] font-semibold tracking-tight text-ink">
   201→            {doc.title}
   202→          </h1>
   203→        </PopIn>
   204→
   205→        <div className="mt-6 flex flex-col items-start gap-5 sm:flex-row sm:items-center">
   206→          <div className="shrink-0">
   207→            <Mascot main={main} size={104} bubbles={false} />
   208→          </div>
   209→          <Reveal delay={0.15} className="min-w-0">
   210→            <SpeechBubble tailClass="left-8 -top-[7px] rotate-[225deg]">
   211→              <p className="text-[14.5px] leading-[1.75] text-ink">
   212→                {doc.intro.map((b, i) =>
   213→                  b.k === "p" ? (
   214→                    <span key={i} className="block">
   215→                      <Fmt text={b.text} />
   216→                    </span>
   217→                  ) : null
   218→                )}
   219→              </p>
   220→            </SpeechBubble>
   221→          </Reveal>
   222→        </div>
   223→
   224→        <PopIn delay={0.18} className="mt-6 flex flex-wrap gap-2">
   225→          {stats.map((s) => (
   226→            <span key={s} className="inline-flex items-center gap-1.5 rounded-full px-3 py-1.5 text-[12px] font-bold text-ink" style={{ background: tint(main, 12) }}>
   227→              <Sparkles className="h-3.5 w-3.5" style={{ color: main }} /> {s}
   228→            </span>
   229→          ))}
   230→        </PopIn>
   231→
   232→        <Squiggle color={main} className="mt-7 h-3.5 w-full opacity-60" />
   233→      </header>
   234→
   235→      {/* ————— contents ————— */}
   236→      <Reveal className="mx-auto max-w-3xl px-4 pt-4 sm:px-8">
   237→        <div className="rounded-[30px] bg-paper p-5 puff sm:p-6">
   238→          <div className="flex items-center gap-3">
   239→            <span className="grid h-10 w-10 shrink-0 place-items-center rounded-2xl" style={{ background: tint(main, 16) }}>
   240→              <Target className="h-5 w-5" style={{ color: inkMix(main, 60) }} />
   241→            </span>
   242→            <div>
   243→              <h2 className="font-display text-[18px] font-semibold text-ink">Contents</h2>
   244→              <p className="text-[12.5px] font-bold text-ink-faint">{doc.sections.length} sections · tap one to jump</p>
   245→            </div>
   246→          </div>
   247→          <div className="mt-4 grid grid-cols-2 gap-2 sm:grid-cols-3">
   248→            {doc.sections.map((s, i) => {
   249→              const accent = accentOf(i);
   250→              return (
   251→                <button
   252→                  key={s.spec}
   253→                  type="button"
   254→                  onClick={() => scrollToSpec(s.spec)}
   255→                  className={`group flex cursor-pointer items-center gap-2 rounded-[18px] px-3 py-2.5 text-left transition-transform hover:-translate-y-0.5${
   256→                    isRoom(i) ? " room" : ""
   257→                  }`}
   258→                  style={{ background: bgOf(i) }}
   259→                >
   260→                  <span
   261→                    className="grid h-6 w-6 shrink-0 place-items-center rounded-full font-display text-[10.5px] font-bold"
   262→                    style={{ background: accent, color: readableInk(accent) }}
   263→                  >
   264→                    {s.spec}
   265→                  </span>
   266→                  <span className="line-clamp-2 text-[12px] font-bold leading-[1.35] text-ink">{s.title}</span>
   267→                </button>
   268→              );
   269→            })}
   270→          </div>
   271→        </div>
   272→      </Reveal>
   273→
   274→      {/* ————— the string of rooms ————— */}
   275→      <div className="mt-10">
   276→        {/* sticky rail rides above every band */}
   277→        <div className="sticky top-[76px] z-40 mx-auto max-w-3xl px-4 sm:px-8">
   278→          <div
   279→            className="rounded-[24px] bg-paper/95 px-3.5 py-2.5 backdrop-blur-md"
   280→            style={{ boxShadow: "0 3px 0 var(--edge), 0 16px 34px -20px rgb(0 0 0 / 0.4)" }}
   281→          >
   282→            <div className="flex items-center gap-3">
   283→              {activeSection ? (
   284→                <span className="min-w-0 flex-1 truncate font-display text-[13.5px] font-semibold text-ink">
   285→                  <span style={{ color: inkMix(accentOf(activeIndex), 45) }}>{activeSection.spec}</span> · {activeSection.title}
   286→                </span>
   287→              ) : (
   288→                <span className="min-w-0 flex-1 truncate font-display text-[13.5px] font-semibold text-ink">{doc.title}</span>
   289→              )}
   290→            </div>
   291→            <div className="relative mt-2">
   292→              <div
   293→                ref={railRef}
   294→                onScroll={measureRail}
   295→                className="no-scrollbar flex items-center gap-1.5 overflow-x-auto"
   296→              >
   297→                {doc.sections.map((s, i) => {
   298→                  const isActive = active === s.spec;
   299→                  const accent = accentOf(i);
   300→                  return (
   301→                    <motion.button
   302→                      key={s.spec}
   303→                      type="button"
   304→                      whileTap={{ scale: 0.88 }}
   305→                      onClick={() => scrollToSpec(s.spec)}
   306→                      className="grid h-8 min-w-8 shrink-0 cursor-pointer place-items-center rounded-full px-2 font-display text-[11px] font-bold"
   307→                      style={
   308→                        isActive
   309→                          ? { background: accent, color: readableInk(accent), boxShadow: `0 3px 0 ${shade(accent)}` }
   310→                          : { background: tint(accent, 22), color: inkMix(accent, 62) }
   311→                      }
   312→                      title={`${s.spec} — ${s.title}`}
   313→                      aria-label={`Jump to spec ${s.spec}`}
   314→                      aria-current={isActive ? "true" : undefined}
   315→                    >
   316→                      {s.spec}
   317→                    </motion.button>
   318→                  );
   319→                })}
   320→              </div>
   321→              {railCut.l && (
   322→                <span
   323→                  aria-hidden
   324→                  className="pointer-events-none absolute inset-y-0 left-0 w-8"
   325→                  style={{ background: "linear-gradient(to right, var(--color-paper) 25%, transparent)" }}
   326→                />
   327→              )}
   328→              {railCut.r && (
   329→                <span
   330→                  aria-hidden
   331→                  className="pointer-events-none absolute inset-y-0 right-0 w-8"
   332→                  style={{ background: "linear-gradient(to left, var(--color-paper) 25%, transparent)" }}
   333→                />
   334→              )}
   335→            </div>
   336→          </div>
   337→        </div>
   338→
   339→        <div className="mt-8">
   340→          <BandWave from="var(--color-cream)" to={bgOf(0)} />
   341→          {doc.sections.map((s, i) => {
   342→            const accent = accentOf(i);
   343→            const room = isRoom(i);
   344→            const last = i === doc.sections.length - 1;
   345→            return (
   346→              <Fragment key={s.spec}>
   347→                <section
   348→                  id={`spec-${s.spec.replace(".", "-")}`}
   349→                  className={`relative scroll-mt-[190px]${room ? " room" : ""}`}
   350→                  style={{ background: bgOf(i) }}
   351→                >
   352→                  <div className="mx-auto max-w-3xl px-4 py-14 sm:px-8 sm:py-16">
   353→                    <SpecSectionView section={s} accent={accent} />
   354→                  </div>
   355→                </section>
   356→                <BandWave from={bgOf(i)} to={last ? "var(--color-cream)" : bgOf(i + 1)} />
   357→              </Fragment>
   358→            );
   359→          })}
   360→        </div>
   361→      </div>
   362→
   363→      {/* ————— end of the note ————— */}
   364→      <div className="mx-auto max-w-3xl px-4 pt-10 pb-24 text-center sm:px-8">
   365→        <Reveal className="flex flex-col items-center gap-5">
   366→          <Squiggle color={main} className="h-3 w-40 opacity-70" />
   367→          <p className="font-display text-[19px] font-semibold text-ink">
   368→            That&apos;s the whole note — every spec point, word for word.
   369→          </p>
   370→          <SoftButton onClick={onBack} ariaLabel="Back to the topic page">
   371→            <ArrowLeft className="h-4.5 w-4.5" /> Back to the topic page
   372→          </SoftButton>
   373→          <p className="mt-2 text-[12px] leading-relaxed font-bold text-ink-soft">
   374→            <Fmt text={`==${unit.code}== · ${unit.title} · text kept verbatim from the ${doc.source}`} />
   375→          </p>
   376→        </Reveal>
   377→      </div>
   378→    </div>
   379→  );
   380→}

   422→/* ————— block dispatcher ————— */
   423→
   424→function BlockView({ block: b, accent }: { block: NotesBlock; accent: string }) {
   425→  switch (b.k) {
   426→    case "p":
   427→      return (
   428→        <p className="max-w-[68ch] text-[16.5px] leading-[1.8] text-ink/90">
   429→          <Fmt text={b.text} />
   430→        </p>
   431→      );
   432→    case "h":
   433→      return (
   434→        <div className="flex items-center gap-2.5 pt-2">
   435→          <span className="h-2.5 w-2.5 shrink-0 rounded-full" style={{ background: accent }} />
   436→          <h3 className="font-display text-[21px] font-semibold tracking-tight text-ink">
   437→            <Fmt text={b.text} />
   438→          </h3>
   439→        </div>
   440→      );
   441→    case "bullets":
   442→      return (
   443→        <ul className="flex max-w-[70ch] flex-col gap-2.5">
   444→          {b.items.map((t, i) => (
   445→            <li key={i} className="flex items-start gap-3 text-[16px] leading-[1.8] text-ink/90">
   446→              <span className="mt-[11px] h-2 w-2 shrink-0 rounded-full" style={{ background: tint(accent, 55) }} />
   447→              <span>
   448→                <Fmt text={t} />
   449→              </span>
   450→            </li>
   451→          ))}
   452→        </ul>
   453→      );
   454→    case "note":
   455→      return (
   456→        <div className="max-w-[72ch] rounded-[20px] px-4 py-3.5" style={{ background: tint("#FFD43B", 18) }}>
   457→          <div className="flex items-start gap-3">
   458→            <StickyNote className="mt-0.5 h-4.5 w-4.5 shrink-0" style={{ color: inkMix("#B08D00", 62) }} />
   459→            <p className="text-[14.5px] leading-[1.7] text-ink">
   460→              <span className="mr-1.5 font-display text-[11.5px] font-extrabold tracking-[0.12em] uppercase" style={{ color: inkMix("#B08D00", 58) }}>{b.label}:</span>
   461→              <Fmt text={b.text} />
   462→            </p>
   463→          </div>
   464→        </div>
   465→      );
   466→    case "def":
   467→      return (
   468→        <div className="max-w-[72ch] rounded-[20px] bg-paper px-5 py-4 puff" style={{ borderLeft: `5px solid ${tint(accent, 40)}` }}>
   469→          <h4 className="font-display text-[16.5px] font-semibold text-ink">
   470→            <Fmt text={b.term} />
   471→          </h4>
   472→          <p className="mt-1.5 text-[15.5px] leading-[1.75] text-ink-soft">
   473→            <Fmt text={b.text} />
   474→          </p>
   475→        </div>
   476→      );
   477→    case "eq":
   478→      return (
   479→        <div className="max-w-[72ch] rounded-[22px] p-5" style={{ background: tint(accent, 9) }}>
   480→          <div className="flex flex-wrap items-center gap-x-3 gap-y-2.5">
   481→            {b.eq.split(/\s{2,}/).map((part, i) => (

   483→                <Fmt text={part.trim()} />
   484→              </span>
   485→            ))}
   486→          </div>
   487→          {b.note && (
   488→            <p className="mt-3 text-[13px] leading-relaxed font-bold text-ink-soft">
   489→              <Fmt text={b.note} />
   490→            </p>
   491→          )}
   492→        </div>
   493→      );
   494→    case "table":
   495→      return <CandyTable head={b.head} rows={b.rows} caption={b.caption} accent={accent} />;
   496→    case "worked":
   497→      return <Worked title={b.title} question={b.question} steps={b.steps} answer={b.answer} accent={accent} />;
   498→    case "steps":
   499→      return <StepsPanel caption={b.caption} items={b.items} accent={accent} />;
   500→    case "checkpoint":
   501→      return (
   502→        <div className="flex items-center gap-3 rounded-full bg-paper px-5 py-4 puff">
   503→          <span className="grid h-10 w-10 shrink-0 place-items-center rounded-full" style={{ background: tint("#12B886", 18) }}>
   504→            <BookOpen className="h-5 w-5 text-mint" />
   505→          </span>
   506→          <p className="text-[15px] font-bold text-ink">
   507→            Checkpoint — the checkpoint questions for this section are on page {b.page} of the textbook.
   508→          </p>
   509→        </div>
   510→      );
   511→    case "figure":
   512→      return <Figure kind={b.kind} caption={b.caption} accent={accent} />;
   513→    case "wow":
   514→      return (
   515→        <div className="max-w-[72ch] rounded-[22px] p-5 sm:p-6" style={{ background: tint("#9775FA", 12) }}>
   516→          <div className="flex items-start gap-3">
   517→            <span className="grid h-9 w-9 shrink-0 place-items-center rounded-2xl bg-paper">
   518→              <Lightbulb className="h-5 w-5" style={{ color: inkMix("#9775FA", 72) }} />
   519→            </span>
   520→            <div>
   521→              <div className="font-display text-[11px] font-extrabold tracking-[0.18em] uppercase" style={{ color: inkMix("#9775FA", 62) }}>Key idea</div>
   522→              <p className="mt-1 text-[15px] leading-[1.75] text-ink">
   523→                <Fmt text={b.text} />
   524→              </p>
   525→            </div>
   526→          </div>
   527→        </div>
   528→      );
   529→    default:
   530→      return null;
   531→  }
   532→}

   683→/* ————— steps — the candy stage strip —————
   684→   One tinted panel, same recipe as the equation block: the caption
   685→   as a header beside a little icon square, and every step as a
   686→   paper mini-card in a loose grid. The accent number chips echo
   687→   the contents grid and the worked-example staircase, so the order
   688→   reads at a glance — no rail, no icons, just tidy cards. */
   689→
   690→function StepsPanel({
   691→  caption,
   692→  items,
   693→  accent,
   694→}: {
   695→  caption?: string;
   696→  items: { title: string; text: string; icon?: string }[];
   697→  accent: string;
   698→}) {
   699→  const n = items.length;
   700→  /* short trios sit side by side; longer tellings wrap into two
   701→     columns, and an odd one out gets the full-width finale */
   702→  const roomy = n >= 4 || items.some((it) => it.text.length > 170);
   703→  const grid = [
   704→    "grid grid-cols-1 gap-2.5 sm:gap-3",
   705→    roomy ? "sm:grid-cols-2" : n === 3 ? "sm:grid-cols-3" : n === 2 ? "sm:grid-cols-2" : "",
   706→    roomy && n % 2 === 1 ? "sm:[&>li:last-child]:col-span-2" : "",
   707→  ]
   708→    .filter(Boolean)
   709→    .join(" ");
   710→
   711→  return (
