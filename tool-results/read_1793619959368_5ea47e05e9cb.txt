    51→/* ═══════════════════════════════════════════════════════════════
    52→   Reaction Club · notes reader — one enclosed room per spec point.
    53→   Each section is a full-width tonal band borrowing one candy
    54→   colour, walking the hue wheel so neighbours always contrast.
    55→   In the hybrid lab every third band is a deep cocoa room with
    56→   milk-toned fonts (see html.hybrid .room in globals.css); in the
    57→   other two labs the same bands stay candy — the class is inert.
    58→   The footer's wave rides between the bands, so the note reads as
    59→   a string of connected rooms rather than one long scroll.
    60→   ═══════════════════════════════════════════════════════════════ */
    61→
    62→/* the rotating palette — hue-wheel order: red → orange → yellow →
    63→   green → purple → pink, so adjacent bands always differ */
    64→const BANDS = ["#FF6B6B", "#FF922B", "#FFD43B", "#12B886", "#9775FA", "#F06595"];
    65→
    66→const WAVE_D = "M0,52 C160,92 340,8 540,30 C740,52 900,96 1100,68 C1260,46 1370,22 1440,40 L1440,90 L0,90 Z";
    67→
    68→function BandWave({ from, to }: { from: string; to: string }) {
    69→  return (
    70→    <div aria-hidden style={{ background: from }}>
    71→      <svg viewBox="0 0 1440 90" preserveAspectRatio="none" className="block h-[46px] w-full sm:h-[68px]">
    72→        {/* fill rides on style so var() strings with fallbacks resolve */}
    73→        <path d={WAVE_D} style={{ fill: to }} />
    74→      </svg>
    75→    </div>
    76→  );
    77→}

    89→/* ————— main reader ————— */
    90→
    91→export function NotesReader({ doc, unit, onBack }: { doc: NotesDoc; unit: Unit; onBack: () => void }) {
    92→  const main = unit.candy.main;
    93→  const [active, setActive] = useState<string>(doc.sections[0]?.spec ?? "");
    94→
    95→  const { scrollYProgress } = useScroll();
    96→  const scaleX = useSpring(scrollYProgress, { stiffness: 140, damping: 26, mass: 0.4 });
    97→
    98→  /* scroll spy */
    99→  const spy = useRef<IntersectionObserver | null>(null);
   100→  useEffect(() => {
   101→    spy.current = new IntersectionObserver(
   102→      (entries) => {
   103→        for (const e of entries) {
   104→          if (e.isIntersecting) setActive(e.target.id.replace("spec-", "").replace("-", "."));
   105→        }
   106→      },
   107→      { rootMargin: "-30% 0px -60% 0px" }
   108→    );
   109→    for (const s of doc.sections) {
   110→      const el = document.getElementById(`spec-${s.spec.replace(".", "-")}`);
   111→      if (el) spy.current?.observe(el);
   112→    }
   113→    return () => spy.current?.disconnect();
   114→  }, [doc]);
   115→
   116→  const count = (k: NotesBlock["k"]) => {
   117→    let n = 0;
   118→    const walk = (blocks: NotesBlock[]) => blocks.forEach((b) => b.k === k && n++);
   119→    walk(doc.intro);
   120→    doc.sections.forEach((s) => walk(s.blocks));
   121→    if (doc.outro) walk(doc.outro);
   122→    return n;
   123→  };
   124→  const stats = [
   125→    `${doc.sections.length} spec sections`,
   126→    `${count("worked")} worked examples`,
   127→    `${count("figure")} diagrams`,
   128→    `${count("checkpoint")} textbook checkpoints`,
   129→  ];
   130→
   131→  /* sticky spec rail — the chip strip scrolls sideways; gradient fades
   132→     hint at the cut-off chips on whichever side is hiding more */
   133→  const railRef = useRef<HTMLDivElement | null>(null);
   134→  const [railCut, setRailCut] = useState({ l: false, r: false });
   135→  const measureRail = () => {
   136→    const el = railRef.current;
   137→    if (!el) return;
   138→    setRailCut({
   139→      l: el.scrollLeft > 4,
   140→      r: el.scrollLeft + el.clientWidth < el.scrollWidth - 4,
   141→    });
   142→  };
   143→  useEffect(() => {
   144→    measureRail();
   145→    window.addEventListener("resize", measureRail);
   146→    return () => window.removeEventListener("resize", measureRail);
   147→    // eslint-disable-next-line react-hooks/exhaustive-deps
   148→  }, [doc]);
   149→
   150→  const accentOf = (i: number) => BANDS[i % BANDS.length];
   151→  /* bands mix their candy over --tint-base at --band-mix — 10% by
   152→     day, 14% after dark, so roast-tone neighbours stay distinct */
   153→  const bandOf = (i: number) =>
   154→    `color-mix(in oklab, ${accentOf(i)} var(--band-mix, 10%), var(--tint-base))`;
   155→  /* every third room goes cocoa in the hybrid lab (2.2, 2.5, 2.8 …).
   156→     Elsewhere --room-surface is undefined and the fallback keeps the
   157→     plain candy band — one expression, correct in all three labs. */
   158→  const isRoom = (i: number) => i % 3 === 1;
   159→  const bgOf = (i: number) => (isRoom(i) ? `var(--room-surface, ${bandOf(i)})` : bandOf(i));
   160→  const activeIndex = Math.max(0, doc.sections.findIndex((s) => s.spec === active));
   161→  const activeSection = doc.sections[activeIndex];
   162→
   163→  return (
   164→    <div className="relative">
   165→      {/* reading progress */}
   166→      <motion.div
   167→        aria-hidden
   168→        className="fixed top-0 left-0 z-[70] h-[7px] w-full origin-left rounded-r-full"
   169→        style={{ scaleX, background: `linear-gradient(90deg, ${main}, #FFD43B)` }}
   170→      />
   171→
   172→      {/* ————— header ————— */}
   173→      <header className="mx-auto max-w-3xl px-4 pt-24 sm:px-8">
   174→        <PopIn>
   175→          <SoftButton onClick={onBack} className="min-h-[42px] px-4 py-2 text-[13.5px]" ariaLabel="Back to the topic page">
   176→            <ArrowLeft className="h-4 w-4" /> Back to the topic page
   177→          </SoftButton>
   178→        </PopIn>
   179→
   180→        <PopIn delay={0.05} className="mt-7 flex flex-wrap items-center gap-2">
   181→          <span
   182→            className="inline-flex -rotate-1 items-center gap-1.5 rounded-full px-3.5 py-1.5 font-display text-[11.5px] font-bold tracking-[0.14em] text-white uppercase"
   183→            style={{ background: main }}
   184→          >
   185→            <BookOpen className="h-3.5 w-3.5" /> Revision note
   186→          </span>
   187→          <span className="inline-flex items-center rounded-full bg-ink/5 px-3.5 py-1.5 text-[12.5px] font-bold capitalize text-ink-soft">
   188→            {doc.source}
   189→          </span>
   190→        </PopIn>
   191→        <PopIn delay={0.1}>
   192→          <p className="mt-6 font-display text-[15px] font-semibold tracking-[0.06em] uppercase" style={{ color: inkMix(main, 52) }}>
   193→            {doc.kicker}
   194→          </p>
   195→          <h1 className="mt-2 font-display text-[clamp(2.1rem,5vw,3.2rem)] leading-[1.05] font-semibold tracking-tight text-ink">
   196→            {doc.title}
   197→          </h1>
   198→        </PopIn>
   199→
   200→        <div className="mt-6 flex flex-col items-start gap-5 sm:flex-row sm:items-center">
   201→          <div className="shrink-0">
   202→            <Mascot main={main} size={104} bubbles={false} />
   203→          </div>
   204→          <Reveal delay={0.15} className="min-w-0">
   205→            <SpeechBubble tailClass="left-8 -top-[7px] rotate-[225deg]">
   206→              <p className="text-[14.5px] leading-[1.75] text-ink">
   207→                {doc.intro.map((b, i) =>
   208→                  b.k === "p" ? (
   209→                    <span key={i} className="block">
   210→                      <Fmt text={b.text} />
   211→                    </span>
   212→                  ) : null
   213→                )}
   214→              </p>
   215→            </SpeechBubble>
   216→          </Reveal>
   217→        </div>
   218→
   219→        <PopIn delay={0.18} className="mt-6 flex flex-wrap gap-2">
   220→          {stats.map((s) => (
   221→            <span key={s} className="inline-flex items-center gap-1.5 rounded-full px-3 py-1.5 text-[12px] font-bold text-ink" style={{ background: tint(main, 12) }}>
   222→              <Sparkles className="h-3.5 w-3.5" style={{ color: main }} /> {s}
   223→            </span>
   224→          ))}
   225→        </PopIn>
   226→
   227→        <Squiggle color={main} className="mt-7 h-3.5 w-full opacity-60" />
   228→      </header>
   229→
   230→      {/* ————— contents ————— */}
   231→      <Reveal className="mx-auto max-w-3xl px-4 pt-4 sm:px-8">
   232→        <div className="rounded-[30px] bg-paper p-5 puff sm:p-6">
   233→          <div className="flex items-center gap-3">
   234→            <span className="grid h-10 w-10 shrink-0 place-items-center rounded-2xl" style={{ background: tint(main, 16) }}>
   235→              <Target className="h-5 w-5" style={{ color: inkMix(main, 60) }} />
   236→            </span>
   237→            <div>
   238→              <h2 className="font-display text-[18px] font-semibold text-ink">Contents</h2>
   239→              <p className="text-[12.5px] font-bold text-ink-faint">{doc.sections.length} sections · tap one to jump</p>
   240→            </div>
   241→          </div>
   242→          <div className="mt-4 grid grid-cols-2 gap-2 sm:grid-cols-3">
   243→            {doc.sections.map((s, i) => {
   244→              const accent = accentOf(i);
   245→              return (
   246→                <button
   247→                  key={s.spec}
   248→                  type="button"
   249→                  onClick={() => scrollToSpec(s.spec)}
   250→                  className={`group flex cursor-pointer items-center gap-2 rounded-[18px] px-3 py-2.5 text-left transition-transform hover:-translate-y-0.5${
   251→                    isRoom(i) ? " room" : ""
   252→                  }`}
   253→                  style={{ background: bgOf(i) }}
   254→                >
   255→                  <span
   256→                    className="grid h-6 w-6 shrink-0 place-items-center rounded-full font-display text-[10.5px] font-bold"
   257→                    style={{ background: accent, color: readableInk(accent) }}
   258→                  >
   259→                    {s.spec}
   260→                  </span>
   261→                  <span className="line-clamp-2 text-[12px] font-bold leading-[1.35] text-ink">{s.title}</span>
   262→                </button>
   263→              );
   264→            })}
   265→          </div>
   266→        </div>
   267→      </Reveal>
   268→
   269→      {/* ————— the string of rooms ————— */}
   270→      <div className="mt-10">
   271→        {/* sticky rail rides above every band */}
   272→        <div className="sticky top-[76px] z-40 mx-auto max-w-3xl px-4 sm:px-8">
   273→          <div
   274→            className="rounded-[24px] bg-paper/95 px-3.5 py-2.5 backdrop-blur-md"
   275→            style={{ boxShadow: "0 3px 0 var(--edge), 0 16px 34px -20px rgb(0 0 0 / 0.4)" }}
   276→          >
   277→            <div className="flex items-center gap-3">
   278→              {activeSection ? (
   279→                <span className="min-w-0 flex-1 truncate font-display text-[13.5px] font-semibold text-ink">
   280→                  <span style={{ color: inkMix(accentOf(activeIndex), 45) }}>{activeSection.spec}</span> · {activeSection.title}
   281→                </span>
   282→              ) : (
   283→                <span className="min-w-0 flex-1 truncate font-display text-[13.5px] font-semibold text-ink">{doc.title}</span>
   284→              )}
   285→            </div>
   286→            <div className="relative mt-2">
   287→              <div
   288→                ref={railRef}
   289→                onScroll={measureRail}
   290→                className="no-scrollbar flex items-center gap-1.5 overflow-x-auto"
   291→              >
   292→                {doc.sections.map((s, i) => {
   293→                  const isActive = active === s.spec;
   294→                  const accent = accentOf(i);
   295→                  return (
   296→                    <motion.button
   297→                      key={s.spec}
   298→                      type="button"
   299→                      whileTap={{ scale: 0.88 }}
   300→                      onClick={() => scrollToSpec(s.spec)}
   301→                      className="grid h-8 min-w-8 shrink-0 cursor-pointer place-items-center rounded-full px-2 font-display text-[11px] font-bold"
   302→                      style={
   303→                        isActive
   304→                          ? { background: accent, color: readableInk(accent), boxShadow: `0 3px 0 ${shade(accent)}` }
   305→                          : { background: tint(accent, 22), color: inkMix(accent, 62) }
   306→                      }
   307→                      title={`${s.spec} — ${s.title}`}
   308→                      aria-label={`Jump to spec ${s.spec}`}
   309→                      aria-current={isActive ? "true" : undefined}
   310→                    >
   311→                      {s.spec}
   312→                    </motion.button>
   313→                  );
   314→                })}
   315→              </div>
   316→              {railCut.l && (
   317→                <span
   318→                  aria-hidden
   319→                  className="pointer-events-none absolute inset-y-0 left-0 w-8"
   320→                  style={{ background: "linear-gradient(to right, var(--color-paper) 25%, transparent)" }}
   321→                />
   322→              )}
   323→              {railCut.r && (
   324→                <span
   325→                  aria-hidden
   326→                  className="pointer-events-none absolute inset-y-0 right-0 w-8"
   327→                  style={{ background: "linear-gradient(to left, var(--color-paper) 25%, transparent)" }}
   328→                />
   329→              )}
   330→            </div>
   331→          </div>
   332→        </div>
   333→
   334→        <div className="mt-8">
   335→          <BandWave from="var(--color-cream)" to={bgOf(0)} />
   336→          {doc.sections.map((s, i) => {
   337→            const accent = accentOf(i);
   338→            const room = isRoom(i);
   339→            const last = i === doc.sections.length - 1;
   340→            return (
   341→              <Fragment key={s.spec}>
   342→                <section
   343→                  id={`spec-${s.spec.replace(".", "-")}`}
   344→                  className={`relative scroll-mt-[190px]${room ? " room" : ""}`}
   345→                  style={{ background: bgOf(i) }}
   346→                >
   347→                  <div className="mx-auto max-w-3xl px-4 py-14 sm:px-8 sm:py-16">
   348→                    <SpecSectionView section={s} accent={accent} />
   349→                  </div>
   350→                </section>
   351→                <BandWave from={bgOf(i)} to={last ? "var(--color-cream)" : bgOf(i + 1)} />
   352→              </Fragment>
   353→            );
   354→          })}
   355→        </div>
   356→      </div>
   357→
   358→      {/* ————— end of the note ————— */}
   359→      <div className="mx-auto max-w-3xl px-4 pt-10 pb-24 text-center sm:px-8">
   360→        <Reveal className="flex flex-col items-center gap-5">
   361→          <Squiggle color={main} className="h-3 w-40 opacity-70" />
   362→          <p className="font-display text-[19px] font-semibold text-ink">
   363→            That&apos;s the whole note — every spec point, word for word.
   364→          </p>
   365→          <SoftButton onClick={onBack} ariaLabel="Back to the topic page">
   366→            <ArrowLeft className="h-4.5 w-4.5" /> Back to the topic page
   367→          </SoftButton>
   368→          <p className="mt-2 text-[12px] leading-relaxed font-bold text-ink-soft">
   369→            <Fmt text={`==${unit.code}== · ${unit.title} · text kept verbatim from the ${doc.source}`} />
   370→          </p>
   371→        </Reveal>
   372→      </div>
   373→    </div>
   374→  );
   375→}

   417→/* ————— block dispatcher ————— */
   418→
   419→function BlockView({ block: b, accent }: { block: NotesBlock; accent: string }) {
   420→  switch (b.k) {
   421→    case "p":
   422→      return (
   423→        <p className="max-w-[68ch] text-[16.5px] leading-[1.8] text-ink/90">
   424→          <Fmt text={b.text} />
   425→        </p>
   426→      );
   427→    case "h":
   428→      return (
   429→        <div className="flex items-center gap-2.5 pt-2">
   430→          <span className="h-2.5 w-2.5 shrink-0 rounded-full" style={{ background: accent }} />
   431→          <h3 className="font-display text-[21px] font-semibold tracking-tight text-ink">
   432→            <Fmt text={b.text} />
   433→          </h3>
   434→        </div>
   435→      );
   436→    case "bullets":
   437→      return (
   438→        <ul className="flex max-w-[70ch] flex-col gap-2.5">
   439→          {b.items.map((t, i) => (
   440→            <li key={i} className="flex items-start gap-3 text-[16px] leading-[1.8] text-ink/90">
   441→              <span className="mt-[11px] h-2 w-2 shrink-0 rounded-full" style={{ background: tint(accent, 55) }} />
   442→              <span>
   443→                <Fmt text={t} />
   444→              </span>
   445→            </li>
   446→          ))}
   447→        </ul>
   448→      );
   449→    case "note":
   450→      return (
   451→        <div className="max-w-[72ch] rounded-[20px] px-4 py-3.5" style={{ background: tint("#FFD43B", 18) }}>
   452→          <div className="flex items-start gap-3">
   453→            <StickyNote className="mt-0.5 h-4.5 w-4.5 shrink-0" style={{ color: inkMix("#B08D00", 62) }} />
   454→            <p className="text-[14.5px] leading-[1.7] text-ink">
   455→              <span className="mr-1.5 font-display text-[11.5px] font-extrabold tracking-[0.12em] uppercase" style={{ color: inkMix("#B08D00", 58) }}>{b.label}:</span>
   456→              <Fmt text={b.text} />
   457→            </p>
   458→          </div>
   459→        </div>
   460→      );
   461→    case "def":
   462→      return (
   463→        <div className="max-w-[72ch] rounded-[20px] bg-paper px-5 py-4 puff" style={{ borderLeft: `5px solid ${tint(accent, 40)}` }}>
   464→          <h4 className="font-display text-[16.5px] font-semibold text-ink">
   465→            <Fmt text={b.term} />
   466→          </h4>
   467→          <p className="mt-1.5 text-[15.5px] leading-[1.75] text-ink-soft">
   468→            <Fmt text={b.text} />
   469→          </p>
   470→        </div>
   471→      );
   472→    case "eq":
   473→      return (
   474→        <div className="max-w-[72ch] rounded-[22px] p-5" style={{ background: tint(accent, 9) }}>
   475→          <div className="flex flex-wrap items-center gap-x-3 gap-y-2.5">
   476→            {b.eq.split(/\s{2,}/).map((part, i) => (
   477→              <span key={i} className="rounded-2xl bg-paper px-4 py-2.5 font-mono text-[14.5px] font-semibold text-ink shadow-[0_3px_0_color-mix(in_oklab,var(--color-ink)_7%,transparent)]">
   478→                <Fmt text={part.trim()} />
   479→              </span>
   480→            ))}
   481→          </div>
   482→          {b.note && (
   483→            <p className="mt-3 text-[13px] leading-relaxed font-bold text-ink-soft">
   484→              <Fmt text={b.note} />
   485→            </p>
   486→          )}
   487→        </div>
   488→      );
   489→    case "table":
   490→      return <CandyTable head={b.head} rows={b.rows} caption={b.caption} accent={accent} />;
   491→    case "worked":
   492→      return <Worked title={b.title} question={b.question} steps={b.steps} answer={b.answer} accent={accent} />;
   493→    case "steps":
   494→      return <Journey caption={b.caption} items={b.items} accent={accent} />;
   495→    case "checkpoint":
   496→      return (
   497→        <div className="flex items-center gap-3 rounded-full bg-paper px-5 py-4 puff">
   498→          <span className="grid h-10 w-10 shrink-0 place-items-center rounded-full" style={{ background: tint("#12B886", 18) }}>
   499→            <BookOpen className="h-5 w-5 text-mint" />
   500→          </span>
   501→          <p className="text-[15px] font-bold text-ink">
   502→            Checkpoint — the checkpoint questions for this section are on page {b.page} of the textbook.
   503→          </p>
   504→        </div>
   505→      );
   506→    case "figure":
   507→      return <Figure kind={b.kind} caption={b.caption} accent={accent} />;
   508→    case "wow":
   509→      return (
   510→        <div className="max-w-[72ch] rounded-[22px] p-5 sm:p-6" style={{ background: tint("#9775FA", 12) }}>
   511→          <div className="flex items-start gap-3">
   512→            <span className="grid h-9 w-9 shrink-0 place-items-center rounded-2xl bg-paper">
   513→              <Lightbulb className="h-5 w-5" style={{ color: inkMix("#9775FA", 72) }} />
   514→            </span>
   515→            <div>
   516→              <div className="font-display text-[11px] font-extrabold tracking-[0.18em] uppercase" style={{ color: inkMix("#9775FA", 62) }}>Key idea</div>
   517→              <p className="mt-1 text-[15px] leading-[1.75] text-ink">
   518→                <Fmt text={b.text} />
   519→              </p>
   520→            </div>
   521→          </div>
   522→        </div>
   523→      );
   524→    default:
   525→      return null;
   526→  }
   527→}
