     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import { AnimatePresence, motion } from "framer-motion";
     5→import {
     6→  ArrowLeft,
     7→  Check,
     8→  Flame,
     9→  RotateCcw,
    10→  Sparkles,
    11→  Trophy,
    12→  X,
    13→  Zap,
    14→} from "lucide-react";
    15→import type { UnitCode } from "@/lib/spectra/types";
    16→import { DECK, DECK_BY_UNIT, shuffle, type Card } from "@/lib/spectra/deck";
    17→import { UNITS } from "@/lib/spectra/units";
    18→import { recordRun, useBlendStats } from "@/lib/spectra/blend-stats";
    19→import { useRevised } from "@/lib/spectra/progress";
    20→import { Fmt } from "./format";
    21→import {
    22→  Blob,
    23→  Chip,
    24→  Confetti,
    25→  DoodleSparkle,
    26→  Float,
    27→  Mascot,
    28→  PopIn,
    29→  Reveal,
    30→  SectionTag,
    31→  SoftButton,
    32→  SpeechBubble,
    33→  tint,
    34→  shade,
    35→} from "./bits";
    36→
    37→/* ═══════════════════════════════════════════════════════════════
    38→   Reaction Club · BLEND MODE
    39→   Every checkpoint question on the site, thrown into a blender.
    40→   Flip → self-judge → streaks grow → confetti at combos.
    41→   Keyboard: Space flip · → got it · ← not yet
    42→   ═══════════════════════════════════════════════════════════════ */
    43→
    44→type Phase = "idle" | "playing" | "done";
    45→
    46→const MILESTONES = [3, 5, 8, 12, 16, 20];
    47→
    48→const COMBO_QUIPS: Record<number, string> = {
    49→  3: "Three in a row! The blend is warming up.",
    50→  5: "FIVE! Somewhere, an enthalpy diagram just smiled.",
    51→  8: "Eight?! Are you a catalyst? Because things are happening faster now.",
    52→  12: "Twelve. This is a runaway reaction and we are here for it.",
    53→  16: "Sixteen! Honestly, revise me harder.",
    54→  20: "TWENTY. Bubbles has fainted. Passionately.",
    55→};
    56→
    57→export function Blend({
    58→  onBack,
    59→  onOpenTopic,
    60→}: {
    61→  onBack: () => void;
    62→  onOpenTopic: (code: UnitCode, topicId: string) => void;
    63→}) {
    64→  const stats = useBlendStats();
    65→  const revised = useRevised();
    66→
    67→  const [phase, setPhase] = useState<Phase>("idle");
    68→  const [filter, setFilter] = useState<UnitCode | "all">("all");
    69→  const [run, setRun] = useState<{ cards: Card[]; picks: ("hit" | "miss")[] } | null>(null);
    70→
    71→  const pile = useMemo(
    72→    () => (filter === "all" ? DECK : DECK.filter((c) => c.unit === filter)),
    73→    [filter]
    74→  );
    75→
    76→  const start = (size: number) => {
    77→    const picked = shuffle(pile).slice(0, size);
    78→    setFiredMilestone(0);
    79→    setRun({ cards: picked, picks: [] });
    80→    setPhase("playing");
    81→    window.setTimeout(
    82→      () => stageRef.current?.scrollIntoView({ behavior: "smooth", block: "start" }),
    83→      80
    84→    );
    85→  };
    86→
    87→  const [firedMilestone, setFiredMilestone] = useState(0);
    88→  const [confetti, setConfetti] = useState(0);
    89→  const [quip, setQuip] = useState<string | null>(null);
    90→  const quipTimer = useRef<number | undefined>(undefined);
    91→
    92→  /* ————— derived run state ————— */
    93→  const idx = run ? run.picks.length : 0;
    94→  const card = run && idx < run.cards.length ? run.cards[idx] : null;
    95→  const hits = run ? run.picks.filter((p) => p === "hit").length : 0;
    96→
    97→  let streak = 0;
    98→  let bestStreak = 0;
    99→  for (const p of run?.picks ?? []) {
   100→    if (p === "hit") {
   101→      streak++;
   102→      bestStreak = Math.max(bestStreak, streak);
   103→    } else streak = 0;
   104→  }
   105→
   106→  const judge = (verdict: "hit" | "miss") => {
   107→    if (!run || run.picks.length >= run.cards.length) return;
   108→    const picks = [...run.picks, verdict];
   109→    setRun({ ...run, picks });
   110→
   111→    // streak bookkeeping (event-driven — no effects, no cascades)
   112→    let s = 0;
   113→    let bs = 0;
   114→    for (const p of picks) {
   115→      if (p === "hit") bs = Math.max(bs, ++s);
   116→      else s = 0;
   117→    }
   118→    if (bs >= 3) {
   119→      const m = MILESTONES.reduce((acc, x) => (bs >= x ? x : acc), 0);
   120→      if (m > firedMilestone) {
   121→        setFiredMilestone(m);
   122→        setConfetti((c) => c + 1);
   123→        setQuip(COMBO_QUIPS[m] ?? "Unstoppable.");
   124→        window.clearTimeout(quipTimer.current);
   125→        quipTimer.current = window.setTimeout(() => setQuip(null), 3200);
   126→      }
   127→    }
   128→
   129→    // last card → record the run and show results
   130→    if (picks.length === run.cards.length) {
   131→      const h = picks.filter((p) => p === "hit").length;
   132→      recordRun(h, picks.length, bs);
   133→      if (picks.length > 0 && h / picks.length >= 0.7) setConfetti((c) => c + 1);
   134→      setPhase("done");
   135→    }
   136→  };
   137→
   138→  const flipRef = useRef<(() => void) | null>(null);
   139→
   140→  /* ————— keyboard ————— */
   141→  useEffect(() => {
   142→    if (phase !== "playing") return;
   143→    const onKey = (e: KeyboardEvent) => {
   144→      if (e.repeat) return;
   145→      if (e.code === "Space" || e.key === " ") {
   146→        e.preventDefault();
   147→        flipRef.current?.();
   148→      } else if (e.key === "ArrowRight") judge("hit");
   149→      else if (e.key === "ArrowLeft") judge("miss");
   150→    };
   151→    window.addEventListener("keydown", onKey);
   152→    return () => window.removeEventListener("keydown", onKey);
   153→  });
   154→
   155→  const stageRef = useRef<HTMLElement | null>(null);
   156→
   157→  return (
   158→    <div className="relative mx-auto max-w-3xl px-5 pt-24 pb-20 sm:px-8">
   159→      {/* back */}
   160→      <PopIn>
   161→        <SoftButton onClick={onBack} className="min-h-[42px] px-4 py-2 text-[13.5px]">
   162→          <ArrowLeft className="h-4 w-4" /> back to the clubhouse
   163→        </SoftButton>
   164→      </PopIn>
   165→
   166→      {/* ————— header ————— */}
   167→      <header className="mt-8">
   168→        <PopIn delay={0.05}>
   169→          <SectionTag main="#9775FA" tilt={-2}>
   170→            the arcade · blend mode
   171→          </SectionTag>
   172→        </PopIn>
   173→        <PopIn delay={0.1}>
   174→          <h1 className="mt-5 font-display text-[clamp(2.3rem,5vw,3.4rem)] leading-[1.05] font-semibold tracking-tight text-ink">
   175→            Flashcards that{" "}
   176→            <span className="relative inline-block -rotate-2 rounded-[0.35em] bg-grape/15 px-[0.18em]">
   177→              fight back.
   178→            </span>
   179→          </h1>
   180→          <p className="mt-3.5 max-w-[58ch] text-[16.5px] leading-[1.8] text-ink-soft">
   181→            Every checkpoint question on the site, blended into one deck. Flip a card, judge
   182→            yourself honestly (Bubbles will know), and chase your streak.{" "}
   183→            <strong className="font-extrabold text-ink">Space</strong> flips,{" "}
   184→            <strong className="font-extrabold text-ink">← →</strong> judges.
   185→          </p>
   186→        </PopIn>
   187→      </header>
   188→
   189→      {/* ————— scoreboard ————— */}
   190→      <PopIn delay={0.16} className="mt-8 flex flex-wrap gap-2.5">
   191→        <span className="inline-flex items-center gap-2 rounded-full bg-grape/15 px-4 py-2.5 text-[13.5px] font-extrabold text-ink">
   192→          <Trophy className="h-4 w-4 text-grape" /> best streak: {stats.best}
   193→        </span>
   194→        <span className="inline-flex items-center gap-2 rounded-full bg-mint/15 px-4 py-2.5 text-[13.5px] font-extrabold text-ink">
   195→          <Zap className="h-4 w-4 text-mint" /> {stats.blends} blends played
   196→        </span>
   197→        {stats.tries > 0 && (
   198→          <span className="inline-flex items-center gap-2 rounded-full bg-lemon/20 px-4 py-2.5 text-[13.5px] font-extrabold text-ink">
   199→            <Sparkles className="h-4 w-4 text-lemon" />
   200→            {Math.round((stats.hits / stats.tries) * 100)}% all-time hit rate
   201→          </span>
   202→        )}
   203→      </PopIn>
   204→
   205→      {/* ————— IDLE: pick a pile ————— */}
   206→      {phase === "idle" && (
   207→        <section className="mt-10">
   208→          <Reveal>
   209→            <div className="mb-3 flex items-center gap-2 font-display text-[13px] font-bold tracking-[0.16em] text-ink-faint uppercase">
   210→              pick your pile <DoodleSparkle className="h-4 w-4 text-lemon" />
   211→            </div>
   212→            <div className="flex flex-wrap gap-2.5">
   213→              <motion.button
   214→                type="button"
   215→                whileHover={{ y: -2 }}
   216→                whileTap={{ scale: 0.95 }}
   217→                onClick={() => setFilter("all")}
   218→                className="cursor-pointer rounded-full px-4 py-2.5 font-display text-[13.5px] font-bold"
   219→                style={
   220→                  filter === "all"
   221→                    ? { background: "#3d2b1f", color: "#fff", boxShadow: "0 3px 0 #1f150e" }
   222→                    : { background: "rgb(61 43 31 / 0.07)", color: "#8a7261" }
   223→                }
   224→              >
   225→                the whole shelf · {DECK.length}
   226→              </motion.button>
   227→              {UNITS.map((u) => (
   228→                <motion.button
   229→                  key={u.code}
   230→                  type="button"
   231→                  whileHover={{ y: -2 }}
   232→                  whileTap={{ scale: 0.95 }}
   233→                  onClick={() => setFilter(u.code)}
   234→                  disabled={DECK_BY_UNIT[u.code] === 0}
   235→                  className="cursor-pointer rounded-full px-4 py-2.5 font-display text-[13.5px] font-bold disabled:cursor-not-allowed disabled:opacity-40"
   236→                  style={
   237→                    filter === u.code
   238→                      ? { background: u.candy.main, color: "#fff", boxShadow: `0 3px 0 ${shade(u.candy.main)}` }
   239→                      : { background: tint(u.candy.main, 14), color: shade(u.candy.main, 62) }
   240→                  }
   241→                >
   242→                  {u.code.replace("WCH", "")} · {u.candy.name} · {DECK_BY_UNIT[u.code]}
   243→                </motion.button>
   244→              ))}
   245→            </div>
   246→          </Reveal>
   247→
   248→          <Reveal delay={0.1} className="mt-10">
   249→            <div
   250→              className="relative flex flex-col items-center gap-6 overflow-hidden rounded-[40px] p-7 text-center sm:p-10"
   251→              style={{ background: tint("#9775FA", 9) }}
   252→            >
   253→              <Blob color={tint("#9775FA", 20)} className="top-[-20%] right-[-10%] h-[60%] w-[45%]" />
   254→              <Float>
   255→                <Mascot main="#9775FA" mood="cheer" size={150} />
   256→              </Float>
   257→              <div className="relative z-10">
   258→                <h2 className="font-display text-[clamp(1.5rem,3.4vw,2.1rem)] font-semibold text-ink">
   259→                  {pile.length} cards ready to blend
   260→                </h2>
   261→                <p className="mx-auto mt-2 max-w-[46ch] text-[14.5px] leading-relaxed text-ink-soft">
   262→                  Ten cards per blend — quick enough for the bus stop, mean enough to matter.
   263→                  Judged by you, remembered by the jar.
   264→                </p>
   265→                <motion.button
   266→                  type="button"
   267→                  onClick={() => start(Math.min(10, pile.length))}
   268→                  whileHover={{ y: -2 }}
   269→                  whileTap={{ scale: 0.95, y: 2 }}
   270→                  className="candy mt-6 inline-flex min-h-[56px] cursor-pointer items-center gap-2.5 rounded-full px-8 py-4 font-display text-[17px] font-semibold text-white select-none"
   271→                  style={{ "--candy-bg": "#9775FA", "--candy-side": shade("#9775FA") } as React.CSSProperties}
   272→                >
   273→                  <Zap className="h-5 w-5" /> Blend {Math.min(10, pile.length)} cards
   274→                </motion.button>
   275→                {pile.length > 10 && (
   276→                  <div className="mt-4 flex flex-wrap justify-center gap-2">
   277→                    {[15, 25].filter((n) => n <= pile.length).map((n) => (
   278→                      <motion.button
   279→                        key={n}
   280→                        type="button"
   281→                        onClick={() => start(n)}
   282→                        whileHover={{ y: -2 }}
   283→                        whileTap={{ scale: 0.95 }}
   284→                        className="cursor-pointer rounded-full bg-white/80 px-4 py-2 text-[12.5px] font-extrabold text-ink-soft"
   285→                      >
   286→                        go big · {n}
   287→                      </motion.button>
   288→                    ))}
   289→                  </div>
   290→                )}
   291→              </div>
   292→            </div>
   293→          </Reveal>
   294→        </section>
   295→      )}
   296→
   297→      {/* ————— PLAYING: the card ————— */}
   298→      {phase === "playing" && run && card && (
   299→        <section ref={stageRef} className="mt-10 scroll-mt-24">
   300→          {/* progress dots */}
   301→          <div className="mb-5 flex items-center justify-center gap-1.5">
   302→            {run.cards.map((_, i) => (
   303→              <motion.span
   304→                key={i}
   305→                aria-hidden
   306→                animate={{
   307→                  scale: i === idx ? 1.5 : 1,
   308→                  backgroundColor:
   309→                    i < idx
   310→                      ? run.picks[i] === "hit"
   311→                        ? "#12b886"
   312→                        : "#ff6b6b"
   313→                      : i === idx
   314→                        ? "#3d2b1f"
   315→                        : "rgb(61 43 31 / 0.15)",
   316→                }}
   317→                className="h-2.5 w-2.5 rounded-full"
   318→              />
   319→            ))}
   320→          </div>
   321→
   322→          {/* streak meter */}
   323→          <div className="mb-5 flex items-center justify-center gap-2">
   324→            <motion.span
   325→              key={streak}
   326→              initial={{ scale: 0.6, rotate: -8 }}
   327→              animate={{ scale: 1, rotate: 0 }}
   328→              transition={{ type: "spring", stiffness: 400, damping: 12 }}
   329→              className={`inline-flex items-center gap-1.5 rounded-full px-4 py-2 font-display text-[14px] font-bold ${
   330→                streak >= 3 ? "text-white" : "bg-ink/5 text-ink-soft"
   331→              }`}
   332→              style={streak >= 3 ? { background: "#ff6b6b", boxShadow: "0 3px 0 #e8590c" } : undefined}
   333→            >
   334→              <Flame className="h-4 w-4" strokeWidth={2.5} />
   335→              streak {streak}
   336→              {bestStreak >= 3 && streak === bestStreak && streak > 0 && (
   337→                <span className="ml-0.5 text-[11px] opacity-80">· on fire!</span>
   338→              )}
   339→            </motion.span>
   340→            <span className="rounded-full bg-ink/5 px-4 py-2 font-display text-[14px] font-bold text-ink-soft">
   341→              {hits} hit · {run.picks.length - hits} missed
   342→            </span>
   343→          </div>
   344→
   345→          <Flashcard
   346→            key={card.id}
   347→            card={card}
   348→            onFlipRef={flipRef}
   349→            onJudge={judge}
   350→            onOpenTopic={onOpenTopic}
   351→          />
   352→
   353→          {/* combo quip */}
   354→          <AnimatePresence>
   355→            {quip && (
   356→              <motion.div
   357→                initial={{ opacity: 0, y: 16, scale: 0.9 }}
   358→                animate={{ opacity: 1, y: 0, scale: 1 }}
   359→                exit={{ opacity: 0, scale: 0.9 }}
   360→                className="mt-6 flex items-end justify-center gap-2"
   361→              >
   362→                <Mascot main="#FFD43B" mood="cheer" size={72} bubbles={false} />
   363→                <SpeechBubble bg="#FFFDF7" className="mb-3" tailClass="left-[-9px] rotate-[135deg]">
   364→                  <span className="font-display text-[13.5px] font-semibold text-ink">{quip}</span>
   365→                </SpeechBubble>
   366→              </motion.div>
   367→            )}
   368→          </AnimatePresence>
   369→
   370→          <div className="relative flex justify-center">
   371→            <Confetti fire={confetti} />
   372→          </div>
   373→
   374→          <p className="mt-8 text-center text-[12.5px] font-bold text-ink-faint">
   375→            Space to flip · → got it · ← not yet
   376→          </p>
   377→        </section>
   378→      )}
   379→
   380→      {/* ————— DONE: results ————— */}
   381→      {phase === "done" && run && (
   382→        <section className="relative mt-10">
   383→          <Reveal>
   384→            <div
   385→              className="relative overflow-hidden rounded-[40px] p-7 text-center sm:p-10"
   386→              style={{ background: tint("#12B886", 9) }}
   387→            >
   388→              <Blob color={tint("#12B886", 18)} className="bottom-[-25%] left-[-10%] h-[65%] w-[50%]" />
   389→              <div className="relative z-10">
   390→                <Float>
   391→                  <Mascot
   392→                    main="#12B886"
   393→                    mood={hits / run.cards.length >= 0.7 ? "cheer" : "idle"}
   394→                    size={150}
   395→                  />
   396→                </Float>
   397→                <h2 className="mt-4 font-display text-[clamp(1.7rem,3.8vw,2.4rem)] font-semibold text-ink">
   398→                  {hits === run.cards.length
   399→                    ? "A perfect blend!"
   400→                    : hits / run.cards.length >= 0.7
   401→                      ? "Smooth operator."
   402→                      : hits / run.cards.length >= 0.4
   403→                        ? "Chunky, but drinkable."
   404→                        : "That one needed sieving."}
   405→                </h2>
   406→                <p className="mt-2 text-[15px] text-ink-soft">
   407→                  {hits} of {run.cards.length} cards hit · best streak {bestStreak}
   408→                  {bestStreak >= stats.best && bestStreak > 0 && (
   409→                    <span className="ml-1.5 inline-flex items-center gap-1 rounded-full bg-lemon/25 px-2.5 py-0.5 text-[11.5px] font-extrabold text-lemon">
   410→                      <Trophy className="h-3 w-3" /> new record
   411→                    </span>
   412→                  )}
   413→                </p>
   414→
   415→                {/* mini bars */}
   416→                <div className="mx-auto mt-6 flex max-w-sm items-center gap-3">
   417→                  <div className="h-5 flex-1 overflow-hidden rounded-full bg-white/80">
   418→                    <motion.div
   419→                      className="h-full rounded-full bg-mint"
   420→                      initial={{ width: 0 }}
   421→                      animate={{ width: `${(hits / run.cards.length) * 100}%` }}
   422→                      transition={{ type: "spring", stiffness: 110, damping: 18, delay: 0.2 }}
   423→                    />
   424→                  </div>
   425→                  <span className="font-display text-[15px] font-bold text-ink">
   426→                    {Math.round((hits / run.cards.length) * 100)}%
   427→                  </span>
   428→                </div>
   429→
   430→                <div className="mt-8 flex flex-wrap items-center justify-center gap-3">
   431→                  <motion.button
   432→                    type="button"
   433→                    onClick={() => {
   434→                      setFiredMilestone(0);
   435→                      setPhase("idle");
   436→                      setRun(null);
   437→                    }}
   438→                    whileHover={{ y: -2 }}
   439→                    whileTap={{ scale: 0.95, y: 2 }}
   440→                    className="candy inline-flex min-h-[52px] cursor-pointer items-center gap-2 rounded-full px-7 py-3.5 font-display text-[16px] font-semibold text-white select-none"
   441→                    style={{ "--candy-bg": "#12B886", "--candy-side": shade("#12B886") } as React.CSSProperties}
   442→                  >
   443→                    <RotateCcw className="h-4.5 w-4.5" /> Blend again
   444→                  </motion.button>
   445→                  <SoftButton
   446→                    onClick={() => {
   447→                      setPhase("idle");
   448→                      setRun(null);
   449→                    }}
   450→                  >
   451→                    Change the pile
   452→                  </SoftButton>
   453→                </div>
   454→              </div>
   455→            </div>
   456→          </Reveal>
   457→
   458→          {/* the cards you missed */}
   459→          {run.picks.some((p) => p === "miss") && (
   460→            <Reveal delay={0.1} className="mt-8">
   461→              <div className="mb-3 font-display text-[13px] font-bold tracking-[0.16em] text-ink-faint uppercase">
   462→                cards that escaped · chase them down
   463→              </div>
   464→              <div className="flex flex-col gap-3">
   465→                {run.cards.map((c, i) =>
   466→                  run.picks[i] === "miss" ? (
   467→                    <motion.button
   468→                      key={c.id}
   469→                      type="button"
   470→                      onClick={() => onOpenTopic(c.unit, c.topicId)}
   471→                      whileHover={{ y: -3, rotate: 0 }}
   472→                      whileTap={{ scale: 0.98 }}
   473→                      className="flex cursor-pointer items-start gap-3.5 rounded-[26px] bg-paper p-4.5 text-left puff sm:items-center sm:p-5"
   474→                      style={{ transform: `rotate(${i % 2 ? 0.5 : -0.5}deg)` }}
   475→                    >
   476→                      <span
   477→                        className="grid h-10 w-10 shrink-0 place-items-center rounded-2xl"
   478→                        style={{ background: tint(c.unitMain, 16) }}
   479→                      >
   480→                        <X className="h-5 w-5" style={{ color: shade(c.unitMain, 62) }} strokeWidth={3} />
   481→                      </span>
   482→                      <span className="min-w-0">
   483→                        <span className="flex flex-wrap items-center gap-2">
   484→                          <Chip main={c.unitMain}>{c.topicLabel}</Chip>
   485→                          <span className="text-[11.5px] font-extrabold text-ink-faint">
   486→                            {c.unit}
   487→                          </span>
   488→                        </span>
   489→                        <span className="mt-1 block text-[14.5px] leading-snug font-bold text-ink">
   490→                          <Fmt text={c.q} />
   491→                        </span>
   492→                      </span>
   493→                    </motion.button>
   494→                  ) : null
   495→                )}
   496→              </div>
   497→            </Reveal>
   498→          )}
   499→
   500→          {/* revised topics hint */}
   501→          <p className="mt-8 text-center text-[12.5px] font-bold text-ink-faint">
   502→            {revised.length === 0
   503→              ? "Tip: topics you mark revised get their own stickers — the jar is watching."
   504→              : `${revised.length} sticker${revised.length === 1 ? "" : "s"} in the jar so far.`}
   505→          </p>
   506→        </section>
   507→      )}
   508→    </div>
   509→  );
   510→}

   512→/* ————— the flashcard itself: a fat 3D flip ————— */
   513→
   514→function Flashcard({
   515→  card,
   516→  onFlipRef,
   517→  onJudge,
   518→  onOpenTopic,
   519→}: {
   520→  card: Card;
   521→  onFlipRef: React.MutableRefObject<(() => void) | null>;
   522→  onJudge: (v: "hit" | "miss") => void;
   523→  onOpenTopic: (code: UnitCode, topicId: string) => void;
   524→}) {
   525→  const [flipped, setFlipped] = useState(false);
   526→  const [nudge, setNudge] = useState(0);
   527→
   528→  const flip = useCallback(() => {
   529→    setFlipped((f) => {
   530→      if (!f) setNudge((n) => n + 1);
   531→      return !f;
   532→    });
   533→  }, []);
   534→
   535→  useEffect(() => {
   536→    onFlipRef.current = flip;
   537→    return () => {
   538→      onFlipRef.current = null;
   539→    };
   540→  }, [flip, onFlipRef]);
   541→
   542→  return (
   543→    <div className="[perspective:1400px]">
   544→      <motion.div
   545→        className="relative mx-auto w-full max-w-xl [transform-style:preserve-3d]"
   546→        animate={{ rotateY: flipped ? 180 : 0, x: nudge ? [0, -6, 6, -3, 0] : 0 }}
   547→        transition={{
   548→          rotateY: { type: "spring", stiffness: 260, damping: 22 },
   549→          x: { duration: 0.4 },
   550→        }}
   551→        style={{ minHeight: 340 }}
   552→      >
   553→        {/* front — the question */}
   554→        <CardFace>
   555→          <div className="flex flex-wrap items-center gap-2">
   556→            <Chip main={card.unitMain}>{card.unit}</Chip>
   557→            <Chip>{card.topicLabel}</Chip>
   558→            <span className="ml-auto font-display text-[12px] font-bold tracking-[0.14em] text-ink-faint uppercase">
   559→              tap to flip
   560→            </span>
   561→          </div>
   562→          <p className="mt-7 font-display text-[clamp(1.3rem,3vw,1.7rem)] leading-[1.4] font-semibold text-ink">
   563→            <Fmt text={card.q} />
   564→          </p>
   565→          <div className="mt-auto pt-7">
   566→            <motion.button
   567→              type="button"
   568→              onClick={flip}
   569→              whileHover={{ y: -2 }}
   570→              whileTap={{ scale: 0.95, y: 2 }}
   571→              className="candy inline-flex min-h-[48px] cursor-pointer items-center gap-2 rounded-full px-6 py-3 font-display text-[15px] font-semibold text-white select-none"
   572→              style={
   573→                {
   574→                  "--candy-bg": card.unitMain,
   575→                  "--candy-side": shade(card.unitMain),
   576→                } as React.CSSProperties
   577→              }
   578→            >
   579→              <Sparkles className="h-4.5 w-4.5" /> Reveal the answer
   580→            </motion.button>
   581→          </div>
   582→        </CardFace>
   583→
   584→        {/* back — the answer */}
   585→        <CardFace back>
   586→          <div className="flex flex-wrap items-center gap-2">
   587→            <span className="inline-flex items-center gap-1.5 rounded-full bg-white px-3.5 py-1.5 font-display text-[12px] font-bold text-ink">
   588→              <Check className="h-3.5 w-3.5 text-mint" strokeWidth={3} /> answer
   589→            </span>
   590→            <span className="text-[12px] font-extrabold text-ink-faint">
   591→              {card.topicTitle}
   592→            </span>
   593→          </div>
   594→          <p className="mt-6 text-[15.5px] leading-[1.8] text-ink">
   595→            <Fmt text={card.a} />
   596→          </p>
   597→          <div className="mt-auto flex flex-col gap-3 pt-7">
   598→            <div className="flex flex-wrap gap-3">
   599→              <motion.button
   600→                type="button"
   601→                onClick={() => onJudge("hit")}
   602→                whileHover={{ y: -2 }}
   603→                whileTap={{ scale: 0.95, y: 2 }}
   604→                className="candy inline-flex min-h-[48px] flex-1 cursor-pointer items-center justify-center gap-2 rounded-full px-5 py-3 font-display text-[15px] font-semibold text-white select-none"
   605→                style={{ "--candy-bg": "#12B886", "--candy-side": "#0ca678" } as React.CSSProperties}
   606→              >
   607→                <Check className="h-4.5 w-4.5" strokeWidth={3} /> Got it
   608→              </motion.button>
   609→              <motion.button
   610→                type="button"
   611→                onClick={() => onJudge("miss")}
   612→                whileHover={{ y: -2 }}
   613→                whileTap={{ scale: 0.95, y: 2 }}
   614→                className="candy inline-flex min-h-[48px] flex-1 cursor-pointer items-center justify-center gap-2 rounded-full px-5 py-3 font-display text-[15px] font-semibold text-white select-none"
   615→                style={{ "--candy-bg": "#FF6B6B", "--candy-side": "#e8590c" } as React.CSSProperties}
   616→              >
   617→                <X className="h-4.5 w-4.5" strokeWidth={3} /> Not yet
   618→              </motion.button>
   619→            </div>
   620→            <button
   621→              type="button"
   622→              onClick={() => onOpenTopic(card.unit, card.topicId)}
   623→              className="cursor-pointer text-[12.5px] font-extrabold text-ink-faint underline decoration-dotted underline-offset-4 hover:text-ink-soft"
   624→            >
   625→              reread {card.topicLabel.toLowerCase()} · {card.topicTitle}
   626→            </button>
   627→          </div>
   628→        </CardFace>
   629→      </motion.div>
   630→
   631→      {/* the whole card is clickable on the front */}
   632→      {!flipped && (
   633→        <button
   634→          type="button"
   635→          aria-label="Flip card"
   636→          onClick={flip}
   637→          className="absolute inset-0 cursor-pointer rounded-[36px]"
   638→          style={{ zIndex: 5 }}
   639→        />
   640→      )}
   641→    </div>
   642→  );
   643→}
