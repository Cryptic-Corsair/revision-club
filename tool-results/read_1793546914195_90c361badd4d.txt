    17→    17→import { UNITS } from "@/lib/spectra/units";

    19→    19→import { useRevised } from "@/lib/spectra/progress";

    37→    37→/* ═══════════════════════════════════════════════════════════════
    38→    38→   Reaction Club · BLEND MODE
    39→    39→   Every checkpoint question on the site, thrown into a blender.
    40→    40→   Flip → self-judge → streaks grow → confetti at combos.
    41→    41→   Keyboard: Space flip · → got it · ← not yet
    42→    42→   ═══════════════════════════════════════════════════════════════ */
    43→    43→
    44→    44→type Phase = "idle" | "playing" | "done";
    45→    45→
    46→    46→const MILESTONES = [3, 5, 8, 12, 16, 20];
    47→    47→
    48→    48→const COMBO_QUIPS: Record<number, string> = {
    49→    49→  3: "Three in a row.",
    50→    50→  5: "Five in a row — keep going.",
    51→    51→  8: "Eight in a row.",
    52→    52→  12: "Twelve in a row.",
    53→    53→  16: "Sixteen in a row.",
    54→    54→  20: "Twenty in a row. Excellent.",
    55→    55→};
    56→    56→
    57→    57→export function Blend({
    58→    58→  onBack,
    59→    59→  onOpenTopic,
    60→    60→}: {
    61→    61→  onBack: () => void;
    62→    62→  onOpenTopic: (code: UnitCode, topicId: string) => void;
    63→    63→}) {
    64→    64→  const stats = useBlendStats();
    65→    65→  const revised = useRevised();
    66→    66→
    67→    67→  const [phase, setPhase] = useState<Phase>("idle");
    68→    68→  const [filter, setFilter] = useState<UnitCode | "all">("all");
    69→    69→  const [run, setRun] = useState<{ cards: Card[]; picks: ("hit" | "miss")[] } | null>(null);
    70→    70→
    71→    71→  const pile = useMemo(
    72→    72→    () => (filter === "all" ? DECK : DECK.filter((c) => c.unit === filter)),
    73→    73→    [filter]
    74→    74→  );
    75→    75→
    76→    76→  const start = (size: number) => {
    77→    77→    const picked = shuffle(pile).slice(0, size);
    78→    78→    setFiredMilestone(0);
    79→    79→    setRun({ cards: picked, picks: [] });
    80→    80→    setPhase("playing");
    81→    81→    window.setTimeout(
    82→    82→      () => stageRef.current?.scrollIntoView({ behavior: "smooth", block: "start" }),
    83→    83→      80
    84→    84→    );
    85→    85→  };
    86→    86→
    87→    87→  const [firedMilestone, setFiredMilestone] = useState(0);
    88→    88→  const [confetti, setConfetti] = useState(0);
    89→    89→  const [quip, setQuip] = useState<string | null>(null);
    90→    90→  const quipTimer = useRef<number | undefined>(undefined);
    91→    91→
    92→    92→  /* ————— derived run state ————— */
    93→    93→  const idx = run ? run.picks.length : 0;
    94→    94→  const card = run && idx < run.cards.length ? run.cards[idx] : null;
    95→    95→  const hits = run ? run.picks.filter((p) => p === "hit").length : 0;
    96→    96→
    97→    97→  let streak = 0;
    98→    98→  let bestStreak = 0;
    99→    99→  for (const p of run?.picks ?? []) {
   100→   100→    if (p === "hit") {
   101→   101→      streak++;
   102→   102→      bestStreak = Math.max(bestStreak, streak);
   103→   103→    } else streak = 0;
   104→   104→  }
   105→   105→
   106→   106→  const judge = (verdict: "hit" | "miss") => {
   107→   107→    if (!run || run.picks.length >= run.cards.length) return;
   108→   108→    const picks = [...run.picks, verdict];
   109→   109→    setRun({ ...run, picks });
   110→   110→
   111→   111→    // streak bookkeeping (event-driven — no effects, no cascades)
   112→   112→    let s = 0;
   113→   113→    let bs = 0;
   114→   114→    for (const p of picks) {
   115→   115→      if (p === "hit") bs = Math.max(bs, ++s);
   116→   116→      else s = 0;
   117→   117→    }
   118→   118→    if (bs >= 3) {
   119→   119→      const m = MILESTONES.reduce((acc, x) => (bs >= x ? x : acc), 0);
   120→   120→      if (m > firedMilestone) {
   121→   121→        setFiredMilestone(m);
   122→   122→        setConfetti((c) => c + 1);
   123→   123→        setQuip(COMBO_QUIPS[m] ?? "Unstoppable.");
   124→   124→        window.clearTimeout(quipTimer.current);
   125→   125→        quipTimer.current = window.setTimeout(() => setQuip(null), 3200);
   126→   126→      }
   127→   127→    }
   128→   128→
   129→   129→    // last card → record the run and show results
   130→   130→    if (picks.length === run.cards.length) {
   131→   131→      const h = picks.filter((p) => p === "hit").length;
   132→   132→      recordRun(h, picks.length, bs);
   133→   133→      if (picks.length > 0 && h / picks.length >= 0.7) setConfetti((c) => c + 1);
   134→   134→      setPhase("done");
   135→   135→    }
   136→   136→  };
   137→   137→
   138→   138→  const flipRef = useRef<(() => void) | null>(null);
   139→   139→
   140→   140→  /* ————— keyboard ————— */
   141→   141→  useEffect(() => {
   142→   142→    if (phase !== "playing") return;
   143→   143→    const onKey = (e: KeyboardEvent) => {
   144→   144→      if (e.repeat) return;
   145→   145→      if (e.code === "Space" || e.key === " ") {
   146→   146→        e.preventDefault();
   147→   147→        flipRef.current?.();
   148→   148→      } else if (e.key === "ArrowRight") judge("hit");
   149→   149→      else if (e.key === "ArrowLeft") judge("miss");
   150→   150→    };
   151→   151→    window.addEventListener("keydown", onKey);
   152→   152→    return () => window.removeEventListener("keydown", onKey);
   153→   153→  });
   154→   154→
   155→   155→  const stageRef = useRef<HTMLElement | null>(null);
   156→   156→
   157→   157→  return (
   158→   158→    <div className="relative mx-auto max-w-3xl px-5 pt-24 pb-20 sm:px-8">
   159→   159→      {/* back */}
   160→   160→      <PopIn>
   161→   161→        <SoftButton onClick={onBack} className="min-h-[42px] px-4 py-2 text-[13.5px]">
   162→   162→          <ArrowLeft className="h-4 w-4" /> back to all units
   163→   163→        </SoftButton>
   164→   164→      </PopIn>
   165→   165→
   166→   166→      {/* ————— header ————— */}
   167→   167→      <header className="mt-8">
   168→   168→        <PopIn delay={0.05}>
   169→   169→          <SectionTag main="#9775FA" tilt={-2}>
   170→   170→            flashcards
   171→   171→          </SectionTag>
   172→   172→        </PopIn>
   173→   173→        <PopIn delay={0.1}>
   174→   174→          <h1 className="mt-5 font-display text-[clamp(2.3rem,5vw,3.4rem)] leading-[1.05] font-semibold tracking-tight text-ink">
   175→   175→            Checkpoint questions,{" "}
   176→   176→            <span className="relative inline-block -rotate-2 rounded-[0.35em] bg-grape/15 px-[0.18em]">
   177→   177→              shuffled.
   178→   178→            </span>
   179→   179→          </h1>
   180→   180→          <p className="mt-3.5 max-w-[58ch] text-[16.5px] leading-[1.8] text-ink-soft">
   181→   181→            Every checkpoint question on the site, shuffled into one deck. Flip a card, answer
   182→   182→            it, then reveal and compare. {" "}
   183→   183→            <strong className="font-extrabold text-ink">Space</strong> flips,{" "}
   184→   184→            <strong className="font-extrabold text-ink">← →</strong> mark it.
   185→   185→          </p>
   186→   186→        </PopIn>
   187→   187→      </header>
   188→   188→
   189→   189→      {/* ————— scoreboard ————— */}
   190→   190→      <PopIn delay={0.16} className="mt-8 flex flex-wrap gap-2.5">
   191→   191→        <span className="inline-flex items-center gap-2 rounded-full bg-grape/15 px-4 py-2.5 text-[13.5px] font-extrabold text-ink">
   192→   192→          <Trophy className="h-4 w-4 text-grape" /> best streak: {stats.best}
   193→   193→        </span>
   194→   194→        <span className="inline-flex items-center gap-2 rounded-full bg-mint/15 px-4 py-2.5 text-[13.5px] font-extrabold text-ink">
   195→   195→          <Zap className="h-4 w-4 text-mint" /> {stats.blends} rounds played
   196→   196→        </span>
   197→   197→        {stats.tries > 0 && (
   198→   198→          <span className="inline-flex items-center gap-2 rounded-full bg-lemon/20 px-4 py-2.5 text-[13.5px] font-extrabold text-ink">
   199→   199→            <Sparkles className="h-4 w-4 text-lemon" />
   200→   200→            {Math.round((stats.hits / stats.tries) * 100)}% all-time correct
   201→   201→          </span>
   202→   202→        )}
   203→   203→      </PopIn>
   204→   204→
   205→   205→      {/* ————— IDLE: pick a pile ————— */}
   206→   206→      {phase === "idle" && (
   207→   207→        <section className="mt-10">
   208→   208→          <Reveal>
   209→   209→            <div className="mb-3 flex items-center gap-2 font-display text-[13px] font-bold tracking-[0.16em] text-ink-faint uppercase">
   210→   210→              choose a deck <DoodleSparkle className="h-4 w-4 text-lemon" />
   211→   211→            </div>
   212→   212→            <div className="flex flex-wrap gap-2.5">
   213→   213→              <motion.button
   214→   214→                type="button"
   215→   215→                whileHover={{ y: -2 }}
   216→   216→                whileTap={{ scale: 0.95 }}
   217→   217→                onClick={() => setFilter("all")}
   218→   218→                className="cursor-pointer rounded-full px-4 py-2.5 font-display text-[13.5px] font-bold"
   219→   219→                style={
   220→   220→                  filter === "all"
   221→   221→                    ? { background: "#3d2b1f", color: "#fff", boxShadow: "0 3px 0 #1f150e" }
   222→   222→                    : { background: "rgb(61 43 31 / 0.07)", color: "#8a7261" }
   223→   223→                }
   224→   224→              >
   225→   225→                the whole course · {DECK.length}
   226→   226→              </motion.button>
   227→   227→              {UNITS.map((u) => (
   228→   228→                <motion.button
   229→   229→                  key={u.code}
   230→   230→                  type="button"
   231→   231→                  whileHover={{ y: -2 }}
   232→   232→                  whileTap={{ scale: 0.95 }}
   233→   233→                  onClick={() => setFilter(u.code)}
   234→   234→                  disabled={DECK_BY_UNIT[u.code] === 0}
   235→   235→                  className="cursor-pointer rounded-full px-4 py-2.5 font-display text-[13.5px] font-bold disabled:cursor-not-allowed disabled:opacity-40"
   236→   236→                  style={
   237→   237→                    filter === u.code
   238→   238→                      ? { background: u.candy.main, color: "#fff", boxShadow: `0 3px 0 ${shade(u.candy.main)}` }
   239→   239→                      : { background: tint(u.candy.main, 14), color: shade(u.candy.main, 62) }
   240→   240→                  }
   241→   241→                >
   242→   242→                  {u.code.replace("WCH", "")} · {DECK_BY_UNIT[u.code]} cards
   243→   243→                </motion.button>
   244→   244→              ))}
   245→   245→            </div>
   246→   246→          </Reveal>
   247→   247→
   248→   248→          <Reveal delay={0.1} className="mt-10">
   249→   249→            <div
   250→   250→              className="relative flex flex-col items-center gap-6 overflow-hidden rounded-[40px] p-7 text-center sm:p-10"
   251→   251→              style={{ background: tint("#9775FA", 9) }}
   252→   252→            >
   253→   253→              <Blob color={tint("#9775FA", 20)} className="top-[-20%] right-[-10%] h-[60%] w-[45%]" />
   254→   254→              <Float>
   255→   255→                <Mascot main="#9775FA" mood="cheer" size={150} />
   256→   256→              </Float>
   257→   257→              <div className="relative z-10">
   258→   258→                <h2 className="font-display text-[clamp(1.5rem,3.4vw,2.1rem)] font-semibold text-ink">
   259→   259→                  {pile.length} cards in this deck
   260→   260→                </h2>
   261→   261→                <p className="mx-auto mt-2 max-w-[46ch] text-[14.5px] leading-relaxed text-ink-soft">
   262→   262→                  Ten cards per round, shuffled each time. Answer first, then check the
   263→   263→                  solution.
   264→   264→                </p>
   265→   265→                <motion.button
   266→   266→                  type="button"
   267→   267→                  onClick={() => start(Math.min(10, pile.length))}
   268→   268→                  whileHover={{ y: -2 }}
   269→   269→                  whileTap={{ scale: 0.95, y: 2 }}
   270→   270→                  className="candy mt-6 inline-flex min-h-[56px] cursor-pointer items-center gap-2.5 rounded-full px-8 py-4 font-display text-[17px] font-semibold text-white select-none"
   271→   271→                  style={{ "--candy-bg": "#9775FA", "--candy-side": shade("#9775FA") } as React.CSSProperties}
   272→   272→                >
   273→   273→                  <Zap className="h-5 w-5" /> Shuffle {Math.min(10, pile.length)} cards
   274→   274→                </motion.button>
   275→   275→                {pile.length > 10 && (
   276→   276→                  <div className="mt-4 flex flex-wrap justify-center gap-2">
   277→   277→                    {[15, 25].filter((n) => n <= pile.length).map((n) => (
   278→   278→                      <motion.button
   279→   279→                        key={n}
   280→   280→                        type="button"
   281→   281→                        onClick={() => start(n)}
   282→   282→                        whileHover={{ y: -2 }}
   283→   283→                        whileTap={{ scale: 0.95 }}
   284→   284→                        className="cursor-pointer rounded-full bg-white/80 px-4 py-2 text-[12.5px] font-extrabold text-ink-soft"
   285→   285→                      >
   286→   286→                        {n} cards
   287→   287→                      </motion.button>
   288→   288→                    ))}
   289→   289→                  </div>
   290→   290→                )}
   291→   291→              </div>
   292→   292→            </div>
   293→   293→          </Reveal>
   294→   294→        </section>
   295→   295→      )}
   296→   296→
   297→   297→      {/* ————— PLAYING: the card ————— */}
   298→   298→      {phase === "playing" && run && card && (
   299→   299→        <section ref={stageRef} className="mt-10 scroll-mt-24">
   300→   300→          {/* progress dots */}
   301→   301→          <div className="mb-5 flex items-center justify-center gap-1.5">
   302→   302→            {run.cards.map((_, i) => (
   303→   303→              <motion.span
   304→   304→                key={i}
   305→   305→                aria-hidden
   306→   306→                animate={{
   307→   307→                  scale: i === idx ? 1.5 : 1,
   308→   308→                  backgroundColor:
   309→   309→                    i < idx
   310→   310→                      ? run.picks[i] === "hit"
   311→   311→                        ? "#12b886"
   312→   312→                        : "#ff6b6b"
   313→   313→                      : i === idx
   314→   314→                        ? "#3d2b1f"
   315→   315→                        : "rgb(61 43 31 / 0.15)",
   316→   316→                }}
   317→   317→                className="h-2.5 w-2.5 rounded-full"
   318→   318→              />
   319→   319→            ))}
   320→   320→          </div>
   321→   321→
   322→   322→          {/* streak meter */}
   323→   323→          <div className="mb-5 flex items-center justify-center gap-2">
   324→   324→            <motion.span
   325→   325→              key={streak}
   326→   326→              initial={{ scale: 0.6, rotate: -8 }}
   327→   327→              animate={{ scale: 1, rotate: 0 }}
   328→   328→              transition={{ type: "spring", stiffness: 400, damping: 12 }}
   329→   329→              className={`inline-flex items-center gap-1.5 rounded-full px-4 py-2 font-display text-[14px] font-bold ${
   330→   330→                streak >= 3 ? "text-white" : "bg-ink/5 text-ink-soft"
   331→   331→              }`}
   332→   332→              style={streak >= 3 ? { background: "#ff6b6b", boxShadow: "0 3px 0 #e8590c" } : undefined}
   333→   333→            >
   334→   334→              <Flame className="h-4 w-4" strokeWidth={2.5} />
   335→   335→              streak {streak}
   336→   336→              {bestStreak >= 3 && streak === bestStreak && streak > 0 && (
   337→   337→                <span className="ml-0.5 text-[11px] opacity-80">· best so far</span>
   338→   338→              )}
   339→   339→            </motion.span>
   340→   340→            <span className="rounded-full bg-ink/5 px-4 py-2 font-display text-[14px] font-bold text-ink-soft">
   341→   341→              {hits} correct · {run.picks.length - hits} missed
   342→   342→            </span>
   343→   343→          </div>
   344→   344→
   345→   345→          <Flashcard
   346→   346→            key={card.id}
   347→   347→            card={card}
   348→   348→            onFlipRef={flipRef}
   349→   349→            onJudge={judge}
   350→   350→            onOpenTopic={onOpenTopic}
   351→   351→          />
   352→   352→
   353→   353→          {/* combo quip */}
   354→   354→          <AnimatePresence>
   355→   355→            {quip && (
   356→   356→              <motion.div
   357→   357→                initial={{ opacity: 0, y: 16, scale: 0.9 }}
   358→   358→                animate={{ opacity: 1, y: 0, scale: 1 }}
   359→   359→                exit={{ opacity: 0, scale: 0.9 }}
   360→   360→                className="mt-6 flex items-end justify-center gap-2"
   361→   361→              >
   362→   362→                <Mascot main="#FFD43B" mood="cheer" size={72} bubbles={false} />
   363→   363→                <SpeechBubble bg="#FFFDF7" className="mb-3" tailClass="left-[-9px] rotate-[135deg]">
   364→   364→                  <span className="font-display text-[13.5px] font-semibold text-ink">{quip}</span>
   365→   365→                </SpeechBubble>
   366→   366→              </motion.div>
   367→   367→            )}
   368→   368→          </AnimatePresence>
   369→   369→
   370→   370→          <div className="relative flex justify-center">
   371→   371→            <Confetti fire={confetti} />
   372→   372→          </div>
   373→   373→
   374→   374→          <p className="mt-8 text-center text-[12.5px] font-bold text-ink-faint">
   375→   375→            Space to flip · → got it · ← not yet
   376→   376→          </p>
   377→   377→        </section>
   378→   378→      )}
   379→   379→
   380→   380→      {/* ————— DONE: results ————— */}
   381→   381→      {phase === "done" && run && (
   382→   382→        <section className="relative mt-10">
   383→   383→          <Reveal>
   384→   384→            <div
   385→   385→              className="relative overflow-hidden rounded-[40px] p-7 text-center sm:p-10"
   386→   386→              style={{ background: tint("#12B886", 9) }}
   387→   387→            >
   388→   388→              <Blob color={tint("#12B886", 18)} className="bottom-[-25%] left-[-10%] h-[65%] w-[50%]" />
   389→   389→              <div className="relative z-10">
   390→   390→                <Float>
   391→   391→                  <Mascot
   392→   392→                    main="#12B886"
   393→   393→                    mood={hits / run.cards.length >= 0.7 ? "cheer" : "idle"}
   394→   394→                    size={150}
   395→   395→                  />
   396→   396→                </Float>
   397→   397→                <h2 className="mt-4 font-display text-[clamp(1.7rem,3.8vw,2.4rem)] font-semibold text-ink">
   398→   398→                  {hits === run.cards.length
   399→   399→                    ? "All correct."
   400→   400→                    : hits / run.cards.length >= 0.7
   401→   401→                      ? "Good work."
   402→   402→                      : hits / run.cards.length >= 0.4
   403→   403→                        ? "Keep practising."
   404→   404→                        : "Worth another look."}
   405→   405→                </h2>
   406→   406→                <p className="mt-2 text-[15px] text-ink-soft">
   407→   407→                  {hits} of {run.cards.length} correct · best streak {bestStreak}
   408→   408→                  {bestStreak >= stats.best && bestStreak > 0 && (
   409→   409→                    <span className="ml-1.5 inline-flex items-center gap-1 rounded-full bg-lemon/25 px-2.5 py-0.5 text-[11.5px] font-extrabold text-lemon">
   410→   410→                      <Trophy className="h-3 w-3" /> new record
   411→   411→                    </span>
   412→   412→                  )}
   413→   413→                </p>
   414→   414→
   415→   415→                {/* mini bars */}
   416→   416→                <div className="mx-auto mt-6 flex max-w-sm items-center gap-3">
   417→   417→                  <div className="h-5 flex-1 overflow-hidden rounded-full bg-white/80">
   418→   418→                    <motion.div
   419→   419→                      className="h-full rounded-full bg-mint"
   420→   420→                      initial={{ width: 0 }}
   421→   421→                      animate={{ width: `${(hits / run.cards.length) * 100}%` }}
   422→   422→                      transition={{ type: "spring", stiffness: 110, damping: 18, delay: 0.2 }}

   514→   514→function Flashcard({
